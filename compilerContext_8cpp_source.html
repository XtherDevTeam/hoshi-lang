44 yoi::indexT compilerContext::compileModule(const yoi::wstr &filepath) {
45 yoi::wstr rFilepath;
46 if (filepath != L"builtin") {
47 for (auto &prep : buildConfig->searchPaths) {
48 std::filesystem::path final = prep / std::filesystem::path(filepath);
49 rFilepath = realpath(final.wstring());
50 if(std::filesystem::exists(rFilepath))
51 break;
52 else
53 continue;
54 }
55 } else {
56 return HOSHI_COMPILER_CTX_GLOB_ID_CONST;
57 }
58
59 if (auto path = std::filesystem::path(rFilepath) / "index.hoshi"; std::filesystem::is_directory(rFilepath) && std::filesystem::exists(path)) {
60 return compileModule(path.wstring());
61 }
62
63 try {
64 return modules.getIndex(rFilepath);
65 } catch (const std::out_of_range &e) {
66 auto fp = fopen(wstring2string(rFilepath).c_str(), "r");
67 if (!fp)
68 throw std::runtime_error("invalid filename: " + wstring2string(rFilepath));
69
70 // temporarily add current directory to search path
71 buildConfig->searchPaths.push_back(std::filesystem::path(filepath).parent_path().wstring());
72
73 fseek(fp, 0, SEEK_END);
74 auto size = ftell(fp);
75 fseek(fp, 0, SEEK_SET);
76 auto *a = new std::string(size, 0);
77 fread(a->data(), size, 1, fp);
78 auto *b = new yoi::wstr{yoi::string2wstring(*a)};
79 delete a;
80 fclose(fp);
81
82 yoi::lexer l{std::wstringstream(*b)};
83 l.scan();
84 delete b;
85 auto current_file = __current_file_path;
86 set_current_file_path(rFilepath);
87 hoshiModule *mod;
88 yoi::parse(mod, l);
89 std::shared_ptr<moduleContext> modCtx = std::make_shared<moduleContext>(shared_from_this(), rFilepath, mod);
90 std::shared_ptr<IRModule> irMod = std::make_shared<IRModule>();
91 irMod->modulePath = rFilepath;
92 auto idx = modules.put(rFilepath, modCtx);
93 irMod->identifier = idx;
94 moduleImported[idx] = irMod;
95 std::shared_ptr<visitor> vis = std::make_shared<visitor>(modCtx, irMod, idx);
96 vis->visit();
97
98 astToFinalize.insert(mod);
99 set_current_file_path(current_file);
100
101 // pop current directory from search path
102 buildConfig->searchPaths.pop_back();
103
104 return idx;
105 }
106 }

112 void compilerContext::initializeSharedObjects() {
113 auto builtinModule = std::make_shared<IRModule>();
114
115 moduleImported[HOSHI_COMPILER_CTX_GLOB_ID_CONST] = builtinModule;
116
117 builtinModuleBuilder = std::make_shared<BuiltinModuleBuilder>(builtinModule);
118 builtinModuleBuilder->build();
119 irFFITable = std::make_shared<IRFFITable>();
120
121 lexer l{std::wstringstream(yoi::string2wstring(__yoi_builtin_module_hoshi))};
122 l.scan();
123 hoshiModule *mod;
124 yoi::parse(mod, l);
125 std::shared_ptr<moduleContext> modCtx = std::make_shared<moduleContext>(shared_from_this(), L"builtin", mod);
126 std::shared_ptr<visitor> vis = std::make_shared<visitor>(modCtx, builtinModule, HOSHI_COMPILER_CTX_GLOB_ID_CONST);
127 vis->visit();
128 astToFinalize.insert(mod);
129 }
