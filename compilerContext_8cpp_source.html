44 yoi::indexT compilerContext::compileModule(const yoi::wstr &filepath) {
45 yoi::wstr rFilepath;
46 if (filepath != L"builtin") {
47 for (auto &prep : buildConfig->searchPaths) {
48 std::filesystem::path final = prep / std::filesystem::path(filepath);
49 rFilepath = realpath(final.wstring());
50 if (std::filesystem::exists(rFilepath) && std::filesystem::is_regular_file(rFilepath)) {
51 break;
52 } else if (std::filesystem::exists(rFilepath + L".hoshi") && std::filesystem::is_regular_file(rFilepath + L".hoshi")) {
53 rFilepath += L".hoshi";
54 break;
55 } else if (std::filesystem::exists(rFilepath) && std::filesystem::is_directory(rFilepath) && std::filesystem::exists(std::filesystem::path(rFilepath) / "index.hoshi")) {
56 rFilepath = (std::filesystem::path(rFilepath) / "index.hoshi").wstring();
57 break;
58 } else {
59 continue;
60 }
61 }
62 } else {
63 return HOSHI_COMPILER_CTX_GLOB_ID_CONST;
64 }
65
66 try {
67 return modules.getIndex(rFilepath);
68 } catch (const std::out_of_range &e) {
69 auto fp = fopen(wstring2string(rFilepath).c_str(), "rb");
70 if (!fp)
71 throw std::runtime_error("invalid filename: " + wstring2string(rFilepath));
72
73 // temporarily add current directory to search path
74 buildConfig->searchPaths.push_back(std::filesystem::path(rFilepath).parent_path().wstring());
75
76 fseek(fp, 0, SEEK_END);
77 auto size = ftell(fp);
78 fseek(fp, 0, SEEK_SET);
79 auto *a = new std::string(size, 0);
80 fread(a->data(), size, 1, fp);
81 auto *b = new yoi::wstr{yoi::string2wstring(*a)};
82 delete a;
83 fclose(fp);
84
85 yoi::lexer l{std::wstringstream(*b)};
86 l.scan();
87 delete b;
88 auto current_file = __current_file_path;
89 set_current_file_path(rFilepath);
90 hoshiModule *mod;
91 yoi::parse(mod, l);
92 std::shared_ptr<moduleContext> modCtx = std::make_shared<moduleContext>(shared_from_this(), rFilepath, mod);
93 std::shared_ptr<IRModule> irMod = std::make_shared<IRModule>();
94 irMod->modulePath = rFilepath;
95 auto idx = modules.put(rFilepath, modCtx);
96 irMod->identifier = idx;
97 moduleImported[idx] = irMod;
98 std::shared_ptr<visitor> vis = std::make_shared<visitor>(modCtx, irMod, idx);
99 vis->visit();
100
101 astToFinalize.insert(mod);
102 set_current_file_path(current_file);
103
104 // pop current directory from search path
105 buildConfig->searchPaths.pop_back();
106
107 return idx;
108 }
109 }

115 void compilerContext::initializeSharedObjects() {
116 auto builtinModule = std::make_shared<IRModule>();
117
118 moduleImported[HOSHI_COMPILER_CTX_GLOB_ID_CONST] = builtinModule;
119
120 builtinModuleBuilder = std::make_shared<BuiltinModuleBuilder>(builtinModule);
121 builtinModuleBuilder->build();
122 irFFITable = std::make_shared<IRFFITable>();
123
124 lexer l{std::wstringstream(yoi::string2wstring(__yoi_builtin_module_hoshi))};
125 l.scan();
126 hoshiModule *mod;
127 yoi::parse(mod, l);
128 builtinModuleContext = std::make_shared<moduleContext>(shared_from_this(), L"builtin", mod);
129 std::shared_ptr<visitor> vis = std::make_shared<visitor>(builtinModuleContext, builtinModule, HOSHI_COMPILER_CTX_GLOB_ID_CONST);
130 vis->visit();
131 astToFinalize.insert(mod);
132 }

207 yoi::IRValueType compilerContext::normalizeForeignBasicType(const std::shared_ptr<yoi::IRValueType> &type, bool handlePointer) {
208 if (type->isForeignBasicType()) {
209 switch (type->type) {
210 case IRValueType::valueType::foreignInt32Type:
211 return *getIntObjectType();
212 case IRValueType::valueType::pointer:
213 return handlePointer ? *getUnsignedObjectType() : *type;
214 case IRValueType::valueType::foreignFloatType:
215 return *getDeciObjectType();
216 default:
217 throw std::runtime_error("unknown foreign basic type");
218 }
219 }
220 return *type;
221 }
