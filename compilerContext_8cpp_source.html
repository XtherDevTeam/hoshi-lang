43 yoi::indexT compilerContext::compileModule(const yoi::wstr &filepath) {
44 yoi::wstr rFilepath;
45 for (auto &prep : buildConfig->searchPaths) {
46 try {
47 std::filesystem::path final = prep / std::filesystem::path(filepath);
48 rFilepath = realpath(final.wstring());
49 break;
50 } catch (std::runtime_error &e) {
51 continue;
52 }
53 }
54 try {
55 return modules.getIndex(rFilepath);
56 } catch (const std::out_of_range &e) {
57 auto fp = fopen(wstring2string(rFilepath).c_str(), "r");
58 if (!fp)
59 throw std::runtime_error("invalid filename: " + wstring2string(rFilepath));
60 fseek(fp, 0, SEEK_END);
61 auto size = ftell(fp);
62 fseek(fp, 0, SEEK_SET);
63 auto *a = new std::string(size, 0);
64 fread(a->data(), size, 1, fp);
65 auto *b = new yoi::wstr{yoi::string2wstring(*a)};
66 delete a;
67 fclose(fp);
68
69 yoi::lexer l{std::wstringstream(*b)};
70 l.scan();
71 delete b;
72 hoshiModule *mod;
73 yoi::parse(mod, l);
74 std::shared_ptr<moduleContext> modCtx = std::make_shared<moduleContext>(shared_from_this(), rFilepath, mod);
75 std::shared_ptr<IRModule> irMod = std::make_shared<IRModule>();
76 irMod->modulePath = rFilepath;
77 auto idx = modules.put(rFilepath, modCtx);
78 irMod->identifier = idx;
79 moduleImported[idx] = irMod;
80 std::shared_ptr<visitor> vis = std::make_shared<visitor>(modCtx, irMod, idx);
81 vis->visit();
82 for (auto &i : irMod->functionTable) {
83 IROptimizer optimizer{shared_from_this(), irMod};
84 optimizer.setTargetFunction(i.second).doOptimizationForCurrentFunction();
85 }
86 finalizeAST(mod);
87 return idx;
88 }
89 }
