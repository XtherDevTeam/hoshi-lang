49 void parse(identifierWithTypeSpec *&o, lexer &lex) {
50 lex.saveState();
51 identifier *id = nullptr;
52 typeSpec *spec = nullptr;
53
54 lexer::token node_start_token = lex.curToken;
55
56 parse(id, lex);
57 if (!id) {
58 lex.dropState(); // No identifier parsed, so no state to return. Drop it.
59 o = nullptr;
60 return;
61 }
62
63 if (lex.curToken.kind == lexer::token::tokenKind::colon) {
64 lex.scan();
65 } else {
66 lex.returnState();
67 finalizeAST(id);
68 o = nullptr;
69 return;
70 }
71
72 parse(spec, lex);
73 if (!spec) { // If typeSpec parsing fails
74 finalizeAST(id);
75 lex.returnState(); // Backtrack because we couldn't complete the rule
76 o = nullptr;
77 return;
78 }
79
80 lex.dropState();
81 o = new identifierWithTypeSpec{node_start_token, id, spec};
82 }

111 lex.saveState();
112 if (lex.curToken.kind != lexer::token::tokenKind::lessThan) {
113 lex.dropState();

118 lex.scan();

416 void parse(externModuleAccessExpression *&o, lexer &lex) {
417 yoi::vec<identifierWithTemplateArg *> vecA;
418 identifierWithTemplateArg *a = nullptr;
419 lexer::token node_start_token = lex.curToken;
420
421 parse(a, lex);
422 if (!a) {
423 o = nullptr;
424 return;
425 }
426
427 while (true) {
428 vecA.push_back(a); // Add current 'a' to vector
429 if (lex.curToken.kind != lexer::token::tokenKind::dot) {
430 break; // No more dots, end of expression
431 } else if (a->hasTemplateArg()) { // Logic: an identifier with template arg cannot be followed by a dot.
432 finalizeAST_vec(vecA);
433 panic(lex.line, lex.col, "expected identifier (except the last term) in externModuleAccessExpression, found identifier with template arguments followed by '.'");
434 o = nullptr;
435 return;
436 } else {
437 lex.scan(); // Consume the dot
438 }
439 a = nullptr; // Reset 'a' for the next parse call
440 parse(a, lex);
441 if (!a) { // A dot must be followed by another identifier.
442 finalizeAST_vec(vecA);
443 panic(lex.line, lex.col, "expected identifier after `.` in externModuleAccessExpression");
444 o = nullptr;
445 return;
446 }
447 }
448 o = new externModuleAccessExpression{node_start_token, vecA};
449 }

605 // Common pattern for binary expressions (mulExpr, addExpr, shiftExpr, etc.)
606 // Applied to all binary expression parsers below.
607 #define PARSE_BINARY_EXPR(NODE_TYPE, CHILD_TYPE, OPERATORS, ERROR_MSG) \
608 void parse(NODE_TYPE *&o, lexer &lex) { \
609 yoi::vec<CHILD_TYPE *> vecA; \
610 yoi::vec<lexer::token> vecB; \
611 CHILD_TYPE *a = nullptr; \
612 lexer::token node_start_token = lex.curToken; \
613\
614 parse(a, lex); \
615 if (a) { \
616 vecA.push_back(a); \
617 while OPERATORS { \
618 vecB.push_back(lex.curToken); \
619 lex.scan(); \
620 a = nullptr; \
621 parse(a, lex); \
622 if (!a) { \
623 finalizeAST_vec(vecA); \
624 panic(lex.line, lex.col, ERROR_MSG); \
625 o = nullptr; \
626 return; \
627 } \
628 vecA.push_back(a); \
629 } \
630 o = new NODE_TYPE{node_start_token, vecA, vecB}; \
631 } else { \
632 o = nullptr; \
633 } \
634 }

1012 lex.saveState();
1013
1014 parse(second, lex); // Try to parse the struct name first
1015 if (!second) {
1016 lex.dropState(); // Drop the saved state as we failed at the very beginning of the rule
1017 panic(lex.line, lex.col, "expected struct name after `impl`");

1028 panic(lex.line, lex.col, "expected interface name after `:`");
1029 o = nullptr;
1030 return;
1031 }

1033 // No colon means no interface specified, 'first' remains nullptr.

1038 parse(inner, lex);
1039 if (!inner) {
1040
1041 if (first) finalizeAST(first);
1042 finalizeAST(second);

1044 lex.dropState();
1045 panic(lex.line, lex.col, "expected implInner after interface or struct name");
1046 o = nullptr;
