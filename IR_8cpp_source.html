59 case operandType::externVar:

62 return L"short(" + std::to_wstring(value.shortV) + L")";
63 case operandType::unsignedInt:
64 return L"unsigned(" + std::to_wstring(value.unsignedV) + L")";

167 void IRBuilder::basicCast(const std::shared_ptr<IRValueType> &valType, yoi::indexT insertionPoint, bool lhs) {
168 switch (valType->type) {
169 case IRValueType::valueType::integerObject:
170 insert({IR::Opcode::basic_cast_int, {}, currentDebugInfo}, insertionPoint);
171 tempVarStack.at(lhs ? tempVarStack.size() - 2 : tempVarStack.size() - 1) =
172 compilerCtx->getIntObjectType();
173 break;
174 case IRValueType::valueType::decimalObject:
175 insert({IR::Opcode::basic_cast_deci, {}, currentDebugInfo}, insertionPoint);
176 tempVarStack.at(lhs ? tempVarStack.size() - 2 : tempVarStack.size() - 1) =
177 compilerCtx->getDeciObjectType();
178 break;
179 case IRValueType::valueType::characterObject:
180 insert({IR::Opcode::basic_cast_char, {}, currentDebugInfo}, insertionPoint);
181 tempVarStack.at(lhs ? tempVarStack.size() - 2 : tempVarStack.size() - 1) =
182 compilerCtx->getCharObjectType();
183 break;
184 case IRValueType::valueType::booleanObject:
185 insert({IR::Opcode::basic_cast_bool, {}, currentDebugInfo}, insertionPoint);
186 tempVarStack.at(lhs ? tempVarStack.size() - 2 : tempVarStack.size() - 1) =
187 compilerCtx->getBoolObjectType();
188 break;
189 case IRValueType::valueType::pointerObject:
190 insert({IR::Opcode::pointer_cast, {}, currentDebugInfo}, insertionPoint);
191 tempVarStack.at(lhs ? tempVarStack.size() - 2 : tempVarStack.size() - 1) =
192 managedPtr(IRValueType{IRValueType::valueType::pointerObject});
193 break;
194 case IRValueType::valueType::shortObject:
195 insert({IR::Opcode::basic_cast_short, {}, currentDebugInfo}, insertionPoint);
196 tempVarStack.at(lhs ? tempVarStack.size() - 2 : tempVarStack.size() - 1) =
197 compilerCtx->getShortObjectType();
198 break;
199 case IRValueType::valueType::unsignedObject:
200 insert({IR::Opcode::basic_cast_unsigned, {}, currentDebugInfo}, insertionPoint);
201 tempVarStack.at(lhs ? tempVarStack.size() - 2 : tempVarStack.size() - 1) =
202 compilerCtx->getUnsignedObjectType();
203 break;
204 default: {
205 panic(currentDebugInfo.line, currentDebugInfo.column, "Unsupported type for basicCast");
206 break;
207 }
208 }

220 void IRBuilder::arithmeticOp(IR::Opcode op) {
221 // fetch lhs and rhs from tempVarStack
222 auto right = tempVarStack.back();
223 tempVarStack.pop_back();
224 auto left = tempVarStack.back();
225 tempVarStack.pop_back();
226 yoi_assert(left->type == right->type, 0, 0, "Type mismatch in arithmetic operation: " + wstring2string(left->to_string()) + " " + std::string{magic_enum::enum_name<>(op)} + " " + wstring2string(right->to_string()));
227 // push result to tempVarStack
228 switch (op) {
229 case IR::Opcode::add:
230 case IR::Opcode::sub:
231 case IR::Opcode::mul:
232 case IR::Opcode::div:
233 case IR::Opcode::mod:
234 case IR::Opcode::left_shift:
235 case IR::Opcode::right_shift:
236 case IR::Opcode::bitwise_not:
237 case IR::Opcode::bitwise_and:
238 case IR::Opcode::bitwise_or:
239 case IR::Opcode::bitwise_xor: {
240 tempVarStack.emplace_back(left);
241 break;
242 }
243 case IR::Opcode::less_than:
244 case IR::Opcode::greater_than:
245 case IR::Opcode::less_equal:
246 case IR::Opcode::greater_equal:
247 case IR::Opcode::equal:
248 case IR::Opcode::not_equal: {
249 tempVarStack.emplace_back(compilerCtx->getBoolObjectType());
250 break;
251 }
252 default: {
253 panic(currentDebugInfo.line, currentDebugInfo.column, "Unsupported type for arithmetic operation.");
254 break;
255 }
256 }
257 insert({op, {}, currentDebugInfo});
258 }

278 void IRBuilder::pushOp(IR::Opcode op, const yoi::IROperand &constV) {
279 if (op == IR::Opcode::push_null) {
280 tempVarStack.emplace_back(managedPtr(IRValueType(IRValueType::valueType::pointerObject)));
281 } else if (constV.type == IROperand::operandType::integer) {
282 // tempVarStack.push_back()
283 tempVarStack.emplace_back(compilerCtx->getIntObjectType());
284 } else if (constV.type == IROperand::operandType::boolean) {
285 tempVarStack.emplace_back(compilerCtx->getBoolObjectType());
286 } else if (constV.type == IROperand::operandType::character) {
287 tempVarStack.emplace_back(compilerCtx->getCharObjectType());
288 } else if (constV.type == IROperand::operandType::decimal) {
289 tempVarStack.emplace_back(compilerCtx->getDeciObjectType());
290 } else if (constV.type == IROperand::operandType::stringLiteral) {
291 tempVarStack.emplace_back(compilerCtx->getStrObjectType());
292 insert({op, {{IROperand::operandType::index, currentModule->identifier}, constV}, currentDebugInfo});
293 return;
294 } else if (constV.type == IROperand::operandType::shortInt) {
295 tempVarStack.emplace_back(compilerCtx->getShortObjectType());
296 } else if (constV.type == IROperand::operandType::unsignedInt) {
297 tempVarStack.emplace_back(compilerCtx->getUnsignedObjectType());
298 } else {
299 panic(currentDebugInfo.line, currentDebugInfo.column, "Unsupported constant type for pushOp");
300 }
301 insert({op, {constV}, currentDebugInfo});
302 }

304 void IRBuilder::loadOp(IR::Opcode op,
305 const yoi::IROperand &source,
306 const std::shared_ptr<IRValueType> &expectedType,
307 yoi::indexT moduleIndex) {
308 if (op == IR::Opcode::load_member || op == IR::Opcode::load_local) {
309 insert({op, {source}, currentDebugInfo});
310 } else if (op == IR::Opcode::load_element) {
311 tempVarStack.pop_back();
312 tempVarStack.pop_back();
313 insert({op, {}, currentDebugInfo});
314 } else if (moduleIndex == -1) {
315 insert({op, {{IROperand::operandType::index, currentModule->identifier}, source}, currentDebugInfo});
316 } else {
317 insert({op, {{IROperand::operandType::index, moduleIndex}, source}, currentDebugInfo});
318 }
319 tempVarStack.emplace_back(expectedType);
320 }

584 IRFunctionDefinition::IRFunctionDefinition(
585 const yoi::wstr &name,
586 const yoi::vec<std::pair<yoi::wstr, std::shared_ptr<IRValueType>>> &argumentTypes,
587 const std::shared_ptr<IRValueType> &returnType, const yoi::vec<std::shared_ptr<IRCodeBlock>> &codeBlock, const yoi::vec<FunctionAttrs> &attrs, const IRDebugInfo &debugInfo)
588 : name(name), returnType(returnType), variableTable(), codeBlock(), debugInfo(debugInfo), attrs(attrs) {
589 variableTable.createScope();
590 for (auto &i : argumentTypes) {

1066 void IRBuilder::newArrayOp(const std::shared_ptr<IRValueType> &elementType,
1067 const yoi::vec<yoi::indexT> &dimensions) {
1068 IR::Opcode op = IR::Opcode::nop;
1069 yoi::vec<IROperand> operands;
1070 switch (elementType->type) {
1071 case IRValueType::valueType::integerObject:
1072 op = IR::Opcode::new_array_int;
1073 break;
1074 case IRValueType::valueType::booleanObject:
1075 op = IR::Opcode::new_array_bool;
1076 break;
1077 case IRValueType::valueType::decimalObject:
1078 op = IR::Opcode::new_array_deci;
1079 break;
1080 case IRValueType::valueType::stringObject:
1081 op = IR::Opcode::new_array_str;
1082 break;
1083 case IRValueType::valueType::characterObject:
1084 op = IR::Opcode::new_array_char;
1085 break;
1086 case IRValueType::valueType::shortObject:
1087 op = IR::Opcode::new_array_short;
1088 break;
1089 case IRValueType::valueType::unsignedObject:
1090 op = IR::Opcode::new_array_unsigned;
1091 break;
1092 case IRValueType::valueType::structObject:
1093 op = IR::Opcode::new_array_struct;
1094 operands.emplace_back(IROperand::operandType::index, elementType->typeAffiliateModule);
1095 operands.emplace_back(IROperand::operandType::index, elementType->typeIndex);
1096 break;
1097 case IRValueType::valueType::interfaceObject:
1098 op = IR::Opcode::new_array_interface;
1099 operands.emplace_back(IROperand::operandType::index, elementType->typeAffiliateModule);
1100 operands.emplace_back(IROperand::operandType::index, elementType->typeIndex);
1101 break;
1102 default:
1103 panic(currentDebugInfo.line, currentDebugInfo.column, "Unsupported array element type: " + yoi::wstring2string(elementType->to_string()));
1104 break;
1105 }
1106 auto size = 1;
1107 for (auto &dim : dimensions) {
1108 operands.emplace_back(IROperand::operandType::index, dim);
1109 size *= dim;
1110 }
1111 for (yoi::indexT i = 0; i < size; ++i) {
1112 tempVarStack.pop_back();
1113 }
1114 insert(IR{op, operands, currentDebugInfo});
1115 tempVarStack.push_back(managedPtr(elementType->getArrayType(dimensions)));
1116 }

1166 void IRBuilder::typeIdOp(const std::shared_ptr<IRValueType> &type) {
1167 IR::Opcode op;
1168 vec<IROperand> operand;
1169 if (type->isArrayType() || type->isDynamicArrayType()) {
1170 yoi::indexT size = 1;
1171 for (auto &dim : type->dimensions) {
1172 size *= dim;
1173 }
1174
1175 op = IR::Opcode::typeid_object_non_stack;
1176 operand.emplace_back(IROperand::operandType::index, static_cast<yoi::indexT>(type->type));
1177 operand.emplace_back(IROperand::operandType::index, static_cast<yoi::indexT>(type->typeAffiliateModule));
1178 operand.emplace_back(IROperand::operandType::index, type->typeIndex);
1179 operand.emplace_back(IROperand::operandType::index, size);
1180 } else {
1181 switch (type->type) {
1182 case IRValueType::valueType::integerObject:
1183 op = IR::Opcode::push_integer;
1184 operand.emplace_back(IROperand::operandType::index, static_cast<yoi::indexT>(0));
1185 break;
1186 case IRValueType::valueType::booleanObject:
1187 op = IR::Opcode::push_integer;
1188 operand.emplace_back(IROperand::operandType::index, static_cast<yoi::indexT>(2));
1189 break;
1190 case IRValueType::valueType::decimalObject:
1191 op = IR::Opcode::push_integer;
1192 operand.emplace_back(IROperand::operandType::index, static_cast<yoi::indexT>(1));
1193 break;
1194 case IRValueType::valueType::characterObject:
1195 op = IR::Opcode::push_integer;
1196 operand.emplace_back(IROperand::operandType::index, static_cast<yoi::indexT>(3));
1197 break;
1198 case IRValueType::valueType::stringObject:
1199 op = IR::Opcode::push_integer;
1200 operand.emplace_back(IROperand::operandType::index, static_cast<yoi::indexT>(4));
1201 break;
1202 case IRValueType::valueType::structObject:
