19 bool ccObjectLinker::commandExists(const std::string& commandName) {
20 const char* path_env = std::getenv("PATH");
21 if (!path_env) {
22 return false; // PATH not set
23 }
24
25 std::string path_str(path_env);
26 std::istringstream iss(path_str);
27 std::string path_segment;
28
29#ifdef _WIN32
30 const char path_delimiter = ';';
31 const std::string executable_extension = ".exe";
32#else
33 const char path_delimiter = ':';
34 const std::string executable_extension;
35#endif
36
37 // iterate through each segment in PATH
38 while (std::getline(iss, path_segment, path_delimiter)) {
39 std::filesystem::path full_path = std::filesystem::path(path_segment) / (commandName + executable_extension);
40
41 // check if the file exists and is a regular file (i.e., not a directory)
42 if (std::filesystem::exists(full_path) && std::filesystem::is_regular_file(full_path)) {
43 return true;
44 }
45 }
46 return false;
47 }

64 ObjectLinker &ccObjectLinker::link(const yoi::wstr &outputPath) {
65 if (this->getLinkerPath().empty()) {
66 throw std::runtime_error("ccObjectLinker: Linker path not set. Call searchAndSetupLinker() first.");
67 }
68 if (this->getObjectPath().empty()) {
69 throw std::runtime_error("ccObjectLinker: Object file path not set.");
70 }
71
72 std::string command = "\"" + yoi::wstring2string(this->getLinkerPath()) + "\"";
73 command += " \"";
74 command += yoi::wstring2string(this->getObjectPath()) + "\"";
75
76 // add additional linking files
77 for (const auto &file : this->getConfig()->additionalLinkingFiles) {
78 command += " \"" + yoi::wstring2string(file) + "\"";
79 }
80
81 command += " -o \"";
82 command += yoi::wstring2string(outputPath) + "\"";
83
84 // add Elysia runtime path and library
85 if (!this->getElysiaRuntimePath().empty()) {
86 // ensure the elysiaRuntimePath exists as a directory
87 std::filesystem::path elysia_path_fs(this->getElysiaRuntimePath());
88 if (!std::filesystem::exists(elysia_path_fs) || !std::filesystem::is_directory(elysia_path_fs)) {
89 std::string error_msg = "ccObjectLinker: Elysia runtime path does not exist or is not a directory: " +
90 yoi::wstring2string(this->getElysiaRuntimePath());
91 throw std::runtime_error(error_msg);
92 }
93
94 command += " -L\""; // add library search path
95 command += yoi::wstring2string(this->getElysiaRuntimePath()) + "\"";
96 command += " -lelysia_runtime";
97 }
98
99 if (this->getConfig()->buildType == IRBuildConfig::BuildType::library) {
100 command += " -shared"; // build a shared library
101 }
102#ifdef _WIN32
103 replace_all(command, std::string("\""), std::string("\\\""));
104 command = "powershell.exe -Command \"&" + command + "\""; // fuck win32 command line
105#endif
106
107 int result = std::system(command.c_str());
108 if (result != 0) {
109 std::string error_msg = "ccObjectLinker: Linker command failed with exit code " + std::to_string(result) + ". Command: " + command;
110 throw std::runtime_error(error_msg);
111 }
112 return *this;
113 }
