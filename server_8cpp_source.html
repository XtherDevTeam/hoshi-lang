44std::string LspServer::readMessage() {
45 // Read Content-Length header
46 std::string line;
47 int contentLength = 0;
48
49 while (std::getline(std::cin, line)) {
50 // Trim trailing \r
51 if (!line.empty() && line.back() == '\r')
52 line.pop_back();
53
54 if (line.empty()) break; // End of headers
55
56 if (line.rfind("Content-Length:", 0) == 0) {
57 std::string lenStr = line.substr(15);
58 // Trim leading space
59 size_t pos = lenStr.find_first_not_of(" \t");
60 if (pos != std::string::npos) lenStr = lenStr.substr(pos);
61 contentLength = std::stoi(lenStr);
62 }
63 }
64
65 if (contentLength <= 0) return "";
66
67 // Read the JSON body
68 std::string content(contentLength, '\0');
69 std::cin.read(&content[0], contentLength);
70 return content;
71}

110void LspServer::handleMessage(const json &msg) {
111 std::string method = msg.value("method", "");
112 json id = msg.contains("id") ? msg["id"] : json();
113 json params = msg.contains("params") ? msg["params"] : json::object();
114
115 bool isRequest = msg.contains("id");
116
117 if (method == "initialize") {
118 handleInitialize(id, params);
119 } else if (method == "initialized") {
120 handleInitialized(params);
121 } else if (method == "shutdown") {
122 handleShutdown(id);
123 } else if (method == "exit") {
124 handleExit();
125 } else if (method == "textDocument/didOpen") {
126 handleDidOpen(params);
127 } else if (method == "textDocument/didChange") {
128 handleDidChange(params);
129 } else if (method == "textDocument/didClose") {
130 handleDidClose(params);
131 } else if (method == "textDocument/completion") {
132 handleCompletion(id, params);
133 } else if (method == "textDocument/hover") {
134 handleHover(id, params);
135 } else if (method == "textDocument/definition") {
136 handleDefinition(id, params);
137 } else if (method == "textDocument/references") {
138 handleReferences(id, params);
139 } else if (method == "textDocument/documentSymbol") {
140 handleDocumentSymbol(id, params);
141 } else if (isRequest) {
142 // Unhandled request — return method not found
143 sendError(id, -32601, "Method not found: " + method);
144 }
145 // Notifications with no handler are silently ignored
146}

152void LspServer::handleInitialize(const json &id, const json &params) {
153 // Extract workspace root for module search paths
154 if (params.contains("rootUri") && params["rootUri"].is_string()) {
155 workspaceRoot = params["rootUri"].get<std::string>();
156 // Strip file:// prefix
157 if (workspaceRoot.rfind("file://", 0) == 0) {
158 workspaceRoot = workspaceRoot.substr(7);
159 }
160 projectIndex.addSearchPath(workspaceRoot);
161 projectIndex.addSearchPath(workspaceRoot + "/.tsuki_modules");
162 projectIndex.addSearchPath(workspaceRoot + "/lib");
163 }
164
165 // Also check rootPath (older LSP clients)
166 if (params.contains("rootPath") && params["rootPath"].is_string() && workspaceRoot.empty()) {
167 workspaceRoot = params["rootPath"].get<std::string>();
168 projectIndex.addSearchPath(workspaceRoot);
169 projectIndex.addSearchPath(workspaceRoot + "/.tsuki_modules");
170 projectIndex.addSearchPath(workspaceRoot + "/lib");
171 }
172
173 // Mirror the compiler's default include paths (main.cpp line 77):
174 // includeDirs{L"", (whereIsHoshiLang() / ".." / "lib").wstring()}
175 // "" = current working directory
176 projectIndex.addSearchPath(".");
177
178 // Global hoshi library: ~/.hoshi/lib/ (whereIsHoshiLang() returns ~/.hoshi/bin)
179 std::filesystem::path hoshiBinPath(yoi::wstring2string(yoi::whereIsHoshiLang()));
180 std::filesystem::path hoshiLibPath = hoshiBinPath / ".." / "lib";
181 std::error_code ec;
182 // canonical() resolves .. and symlinks for a clean path
183 std::string globalLib = std::filesystem::canonical(hoshiLibPath, ec).string();
184 if (!ec) {
185 projectIndex.addSearchPath(globalLib);
186 projectIndex.addSearchPath(globalLib + "/.tsuki_modules");
187 }
188
189 InitializeResult result;
190 sendResponse(id, json(result));
191}

252void LspServer::publishDiagnostics(const std::string &uri, int version) {
253 Document *doc = documents.getDocument(uri);
254 if (!doc) return;
255
256 PublishDiagnosticsParams pub;
257 pub.uri = uri;
258 pub.version = version;
259
260 for (auto &diag : doc->diagnostics) {
261 LspDiagnostic ld;
262 // Convert 1-based (internal) to 0-based (LSP)
263 ld.range.start.line = static_cast<int>(diag.line) - 1;
264 ld.range.start.character = static_cast<int>(diag.column) - 1;
265 ld.range.end.line = ld.range.start.line;
266 ld.range.end.character = ld.range.start.character + 1;
267
268 switch (diag.severity) {
269 case yoi::DiagnosticSeverity::Error:
270 ld.severity = DiagnosticSeverity::Error;
271 break;
272 case yoi::DiagnosticSeverity::Warning:
273 ld.severity = DiagnosticSeverity::Warning;
274 break;
275 case yoi::DiagnosticSeverity::Note:
276 ld.severity = DiagnosticSeverity::Information;
277 break;
278 case yoi::DiagnosticSeverity::Hint:
279 ld.severity = DiagnosticSeverity::Hint;
280 break;
281 }
282
283 ld.message = diag.message;
284 ld.source = "hoshi-lang";
285 pub.diagnostics.push_back(ld);
286 }
287
288 sendNotification("textDocument/publishDiagnostics", json(pub));
289}

340// Safe cast: yoi::indexT (uint64_t) → int for LSP positions.
341// Clamps to [0, INT_MAX] to avoid negative values from overflow.
342static int safeInt(yoi::indexT v) {
343 return static_cast<int>(v > static_cast<yoi::indexT>(INT_MAX) ? INT_MAX : v);
344}

369void LspServer::handleDocumentSymbol(const json &id, const json &params) {
370 auto uri = params["textDocument"]["uri"].get<std::string>();
371 Document *doc = documents.getDocument(uri);
372
373 std::vector<DocumentSymbol> result;
374
375 if (doc && doc->parseSucceeded) {
376 for (auto &sym : doc->symbols) {
377 DocumentSymbol ds;
378 ds.name = yoi::wstring2string(sym.name);
379 ds.detail = yoi::wstring2string(sym.detail);
380 ds.kind = mapSymbolKind(sym.kind);
381
382 int sl = safeInt(sym.line);
383 int sc = safeInt(sym.column);
384 int el = safeInt(sym.endLine > 0 ? sym.endLine : sym.line);
385 int ec = safeInt(sym.endColumn > 0 ? sym.endColumn : sym.column + sym.name.size());
386 // Ensure end >= start
387 if (ec <= sc) ec = sc + 1;
388 if (el < sl) el = sl;
389
390 ds.range.start.line = sl;
391 ds.range.start.character = sc;
392 ds.range.end.line = el;
393 ds.range.end.character = ec;
394 ds.selectionRange = ds.range;
395
396 for (auto &child : sym.children) {
397 DocumentSymbol cs;
398 cs.name = yoi::wstring2string(child.name);
399 cs.detail = yoi::wstring2string(child.detail);
400 cs.kind = mapSymbolKind(child.kind);
401
402 int cl = safeInt(child.line);
403 int cc = safeInt(child.column);
404 int cel = cl;
405 int cec = cc + safeInt(child.name.size());
406 if (cec <= cc) cec = cc + 1;
407
408 cs.range.start.line = cl;
409 cs.range.start.character = cc;
410 cs.range.end.line = cel;
411 cs.range.end.character = cec;
412 cs.selectionRange = cs.range;
413 ds.children.push_back(cs);
414 }
415
416 result.push_back(ds);
417 }
418 }
419
420 sendResponse(id, json(result));
421}
