14 void parseString(std::wistream &input, wstr &value) {
15 wchar ch = '\0';
16 while (input) {
17 if (!input.get(ch)) break;
18 if (ch == '\\') {
19 if (!input.get(ch)) break;
20 switch (ch) {
21 case '\\':
22 case '"':
23 case '\'':
24 case '/':
25 value.push_back(ch);
26 break;
27 case 'b':
28 value.push_back('\b');
29 break;
30 case 'f':
31 value.push_back('\f');
32 break;
33 case 'n':
34 value.push_back('\n');
35 break;
36 case 'r':
37 value.push_back('\r');
38 break;
39 case 't':
40 value.push_back('\t');
41 break;
42 case 'u': {
43 wchar fuckutf{};
44 for (int64_t i = 3; input && i >= 0; i--) {
45 if (!input.get(ch)) break;
46 if ('a' <= ch and ch <= 'z')
47 fuckutf += ((ch - 'a' + 10) * (1 << 4 * i));
48 else if ('A' <= ch and ch <= 'Z')
49 fuckutf += ((ch - 'A' + 10) * (1 << 4 * i));
50 else
51 fuckutf += ((ch - '0') * (1 << 4 * i));
52 }
53 value.push_back(fuckutf);
54 break;
55 }
56 default:
57 value.push_back(ch);
58 break;
59 }
60 } else {
61 value.push_back(ch);
62 }
63 }
