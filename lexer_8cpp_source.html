12 void lexer::getCh() {
13 start:
14 if (!stream) {
15 throw std::runtime_error("hoshi::lexer::getCh() - eof");
16 }
17 if (!stream.get(curCh) || stream.fail()) {
18 curCh = '\0';
19 }
20 if (curCh == L'\n') {
21 line++, col = 0;
22 } else if (curCh == L'\r') {
23 goto start;
24 } else {
25 col++;
26 }
27 }

29 lexer::token lexer::scan() {
30 if (curCh == '\0') {
31 return curToken = lexer::token {line, col, token::tokenKind::eof};
32 }
33 while (curCh == ' ' or curCh == '\n' or curCh == '\t') getCh();
34 if (std::isalpha(curCh) or curCh == '_') {
35 return curToken = alphaStart();
36 } else if (std::isdigit(curCh)) {
37 return curToken = digitStart();
38 } else if (curCh == L'+') {
39 return curToken = plusStart();
40 } else if (curCh == L'-') {
41 return curToken = minusStart();
42 } else if (curCh == L'*') {
43 return curToken = asteriskStart();
44 } else if (curCh == L'/') {
45 return curToken = slashStart();
46 } else if (curCh == L'%') {
47 return curToken = percentSignStart();
48 } else if (curCh == L'!') {
49 return curToken = notStart();
50 } else if (curCh == L'=') {
51 return curToken = equalStart();
52 } else if (curCh == L'>') {
53 return curToken = greaterStart();
54 } else if (curCh == L'<') {
55 return curToken = lessStart();
56 } else if (curCh == L'"' or curCh == L'\'') {
57 return curToken = strStart();
58 } else if (curCh == L';') {
59 return curToken = semicolonStart();
60 } else if (curCh == L':') {
61 return curToken = colonStart();
62 } else if (curCh == L',') {
63 return curToken = commaStart();
64 } else if (curCh == L'.') {
65 return curToken = dotStart();
66 } else if (curCh == L'(') {
67 return curToken = leftParenthesesStart();
68 } else if (curCh == L')') {
69 return curToken = rightParenthesesStart();
70 } else if (curCh == L'[') {
71 return curToken = leftBracketStart();
72 } else if (curCh == L']') {
73 return curToken = rightBracketStart();
74 } else if (curCh == L'{') {
75 return curToken = leftBracesStart();
76 } else if (curCh == L'}') {
77 return curToken = rightBracesStart();
78 } else if (curCh == L'&') {
79 return curToken = andStart();
80 } else if (curCh == L'|') {
81 return curToken = orStart();
82 } else if (curCh == L'^') {
83 return curToken = xorStart();
84 } else if (curCh == L'#') {
85 return curToken = sharpStart();
86 } else if (curCh == L'!') {
87 return curToken = notStart();
88 } else if (curCh == L'\0') {
89 return curToken = {line, col, token::tokenKind::eof, token::vBasicValue{false}};
90 } else {
91 throw std::runtime_error("hoshi::lexer::scan() - undefined token");
92 }

193 lexer::token lexer::strStart() {
194 wchar strV = curCh;
195 lexer::token tok{line, col, strV == L'"' ? token::tokenKind::string : token::tokenKind::character, wstr()};
196 getCh();
197 while (curCh != strV) {
198 if (curCh == '\\') {
199 getCh();
200 tok.strVal += '\\';
201 }
202 tok.strVal += curCh;
203 getCh();
204 }
205 getCh(); // skip "
206 std::wistringstream ss{tok.strVal};
207 tok.strVal = {};
208 parseString(ss, tok.strVal);
209 if (strV == L'\'' && tok.strVal.size() > 1)
210 throw std::runtime_error("lexer::strStart() - character literal length > 1");
211 return tok;
212 }

214 lexer::token lexer::digitStart() {
215 enum class MatchPattern {
216 hex,
217 oct,
218 dec,
219 bin
220 } matchPattern = MatchPattern::dec;
221 lexer::token tok{line, col, token::tokenKind::integer, token::vBasicValue{(int64_t) 0}};
222 wstr tempStr;
223 if (curCh == '0') {
224 tempStr += curCh;
225 getCh();
226 switch (curCh) {
227 case 'x':
228 case 'X':
229 matchPattern = MatchPattern::hex;
230 getCh();
231 tempStr += curCh;
232 getCh();
233 break;
234 case 'b':
235 case 'B':
236 matchPattern = MatchPattern::bin;
237 getCh();
238 tempStr += curCh;
239 getCh();
240 break;
241 case '0':
242 case '1':
243 case '2':
244 case '3':
245 case '4':
246 case '5':
247 case '6':
248 case '7':
249 matchPattern = MatchPattern::oct;
250 tempStr += curCh;
251 getCh();
252 break;
253 case 'o':
254 case 'O':
255 matchPattern = MatchPattern::oct;
256 getCh();
257 tempStr += curCh;
258 getCh();
259 break;
260 default:
261 break;
262 }
263 }
264 while (isdigit(curCh) || (matchPattern == MatchPattern::hex && isxdigit(curCh))) {
265 tempStr += curCh;
266 getCh();
267 }
268 if (curCh == '.') {
269 yoi_assert(line, col, matchPattern == MatchPattern::dec, "lexer::digitStart() - invalid match pattern");
270 tok.kind = token::tokenKind::decimal;
271 tempStr += curCh;
272 getCh();
273 while (isdigit(curCh)) {
274 tempStr += curCh;
275 getCh();
276 }
277 }
278 if (tok.kind == token::tokenKind::integer) {
279 switch (curCh) {
280 case 'u':
281 case 'U':
282 getCh();
283 tok.kind = token::tokenKind::unsignedInt;
284 break;
285 case 's':
286 case 'S':
287 getCh();
288 tok.kind = token::tokenKind::shortInt;
289 break;
290 }
291 int base = 10;
292 switch (matchPattern) {
293 case MatchPattern::dec: {
294 base = 10;
295 break;
296 }
297 case MatchPattern::hex: {
298 base = 16;
299 break;
300 }
301 case MatchPattern::oct: {
302 base = 8;
303 break;
304 }
305 case MatchPattern::bin: {
306 base = 2;
307 break;
308 }
309 }
310 if (tok.kind == token::tokenKind::integer)
311 tok.basicVal.vInt = std::stoll(tempStr, nullptr, base);
312 else if (tok.kind == token::tokenKind::unsignedInt)
313 tok.basicVal.vUint = std::stoull(tempStr, nullptr, base);
314 else if (tok.kind == token::tokenKind::shortInt)
315 tok.basicVal.vShort = static_cast<int16_t>(std::stoi(tempStr, nullptr, base));
316 } else {
317 tok.basicVal.vDeci = std::stod(tempStr);
318 }
319 return tok;
320 }

361 lexer::token lexer::slashStart() {
362 lexer::token tok{line, col, token::tokenKind::slash};
363 getCh();
364 if (curCh == '=') {
365 tok.kind = token::tokenKind::divisionAssignment;
366 getCh();
367 } else if (curCh == '/') {
368 getCh();
369 while (curCh and curCh != '\n')

378 getCh();
379 break;
380 }

382 getCh();
383 }
384 return scan(); // 多行注释解析

617 lexer::token lexer::operatorStart() {
618 yoi::wstr operatorId{L"operator"};
619 switch (curToken.kind) {
620 case token::tokenKind::plus: operatorId += L"+"; break;
621 case token::tokenKind::minus: operatorId += L"-"; break;
622 case token::tokenKind::asterisk: operatorId += L"*"; break;
623 case token::tokenKind::slash: operatorId += L"/"; break;
624 case token::tokenKind::percentSign: operatorId += L"%"; break;
625 case token::tokenKind::lessThan: operatorId += L"<"; break;
626 case token::tokenKind::greaterThan: operatorId += L">"; break;
627 case token::tokenKind::equal: operatorId += L"=="; break;
628 case token::tokenKind::notEqual: operatorId += L"!="; break;
629 case token::tokenKind::binaryAnd: operatorId += L"&"; break;
630 case token::tokenKind::binaryNot: operatorId += L"~"; break;
631 case token::tokenKind::binaryOr: operatorId += L"|"; break;
632 case token::tokenKind::binaryXor: operatorId += L"^"; break;
633 case token::tokenKind::binaryShiftLeft: operatorId += L"<<"; break;
634 case token::tokenKind::binaryShiftRight: operatorId += L">>"; break;
635 case token::tokenKind::incrementSign: operatorId += L"++"; break;
636 case token::tokenKind::decrementSign: operatorId += L"--"; break;
637 case token::tokenKind::directAssignSign: operatorId += L":="; break;
638 case token::tokenKind::additionAssignment: operatorId += L"+="; break;
639 case token::tokenKind::subtractionAssignment: operatorId += L"-="; break;
640 case token::tokenKind::multiplicationAssignment: operatorId += L"*="; break;
641 case token::tokenKind::divisionAssignment: operatorId += L"/="; break;
642 case token::tokenKind::reminderAssignment: operatorId += L"%="; break;
643 case token::tokenKind::assignSign: operatorId += L"="; break;
644 case token::tokenKind::leftParentheses: {
645 scan();
646 yoi_assert(curToken.kind == token::tokenKind::rightParentheses, line, col, "lexer::operatorStart() - invalid operator");
647 operatorId += L"()";
648 break;
649 }
650 case token::tokenKind::leftBracket: {
651 scan();
652 yoi_assert(curToken.kind == token::tokenKind::rightBracket, line, col, "lexer::operatorStart() - invalid operator");
653 operatorId += L"[]";
654 break;
655 }
656 default: throw std::runtime_error("lexer::operatorStart() - unknown operator");
657 }
658 return curToken = lexer::token{line, col, token::tokenKind::identifier, std::move(operatorId)};
659 }
