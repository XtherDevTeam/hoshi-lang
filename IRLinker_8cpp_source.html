15 std::shared_ptr<IRObjectFile> IRLinker::link(const std::shared_ptr<compilerContext>& context, indexT entryId) {
16 this->compilerCtx = context;
17 this->entryModuleId = entryId;
18 this->finalModule = std::make_shared<IRModule>();
19 finalModule->identifier = ENTRY_MODULE_ID_CONST; // Unified module has no specific ID, elysia here
20
21 auto objectFile = std::make_shared<IRObjectFile>();
22 objectFile->compiledModule = this->finalModule;
23 objectFile->entryModule = this->entryModuleId;
24
25 // The order is important:
26 // 1. Strings, Structs, Interfaces: Define types first.
27 // 2. Globals: Depend on types.
28 // 3. Functions: Depend on everything.
29 linkStringLiterals();
30 linkStructsAndInterfaces();
31 linkGlobals();
32 linkFunctions();
33 linkInterfaceImplementations();
34 patchIRFFITable();
35 createEntryFunction();
36
37 return objectFile;
38 }

40 wstr IRLinker::mangleName(indexT moduleId, const wstr& originalName) {
41 // only when it's not a main function, we need to mangle the name
42 if (moduleId == entryModuleId && originalName == L"main#") {
43 return L"yoi_main";
44 }
45 std::wstringstream ss;
46 ss << std::hex << moduleId;
47 return ss.str() + L"_" + originalName;
48 }

50 void IRLinker::linkStringLiterals() {
51 for (const auto& modPair : compilerCtx->getCompiledModules()) {
52 indexT modId = modPair.first;
53 const auto& srcModule = modPair.second;
54 for (indexT oldIdx = 0; oldIdx < srcModule->stringLiteralPool.pool.size(); ++oldIdx) {
55 const auto& str = srcModule->stringLiteralPool.pool[oldIdx];
56 indexT newIdx = finalModule->stringLiteralPool.addStringLiteral(str);

62 void IRLinker::linkStructsAndInterfaces() {
63 for (const auto& modPair : compilerCtx->getCompiledModules()) {
64 indexT modId = modPair.first;
65 const auto& srcModule = modPair.second;
66
67 // Link Structs
68 for (const auto& structPair : srcModule->structTable) {
69 indexT oldIdx = srcModule->structTable.getIndex(structPair.first);
70 auto newName = mangleName(modId, structPair.second->name);
71 // Just copy the definition, field types will be patched later if needed
72 auto newIndex = finalModule->structTable.put_create(newName, structPair.second);
73 structRemapping[modId][oldIdx] = newIndex;
74 }
75
76 for (const auto& ifacePair : srcModule->interfaceTable) {
77 indexT oldIdx = srcModule->interfaceTable.getIndex(ifacePair.first);
78 auto newName = mangleName(modId, ifacePair.second->name);
79 // Just copy the definition, method types will be patched later if needed
80 finalModule->interfaceTable.put_create(newName, ifacePair.second);
81 auto newIndex = finalModule->interfaceTable.getIndex(newName);
82 interfaceRemapping[modId][oldIdx] = newIndex;
83 }
84 }
85
86 for (const auto& modPair : compilerCtx->getCompiledModules()) {
87 indexT modId = modPair.first;
88 const auto& srcModule = modPair.second;
89
90 for (const auto& structPair : srcModule->structTable) {
91 indexT oldIdx = srcModule->structTable.getIndex(structPair.first);
92 auto newName = mangleName(modId, structPair.second->name);
93 indexT newIdx = finalModule->structTable.getIndex(newName);
94 finalModule->structTable[newIdx]->name = newName;
95 for (auto &type : finalModule->structTable[newIdx]->fieldTypes) {
96 *type = *patchType(type);
97 }
98 }
99
100 // Link Interfaces
101 for (const auto& ifacePair : srcModule->interfaceTable) {
102 indexT oldIdx = srcModule->interfaceTable.getIndex(ifacePair.first);
103 auto newName = mangleName(modId, ifacePair.second->name);
104 indexT newIdx = finalModule->interfaceTable.getIndex(newName);
105 finalModule->interfaceTable[newIdx]->name = newName;
106
107 for (auto &method : finalModule->interfaceTable[newIdx]->methodMap) {
108 for (auto &param : method.second->argumentTypes) {
109 *param = *patchType(param);
110 }
111 for (auto &ret : method.second->variableTable.getVariables()) {
112 *ret = *patchType(ret);
113 }
114 method.second->returnType = patchType(method.second->returnType);
115 }
116
117 interfaceRemapping[modId][oldIdx] = newIdx;
118 }
119
120 // Link Interface Implementations
121 for (const auto& implPair : srcModule->interfaceImplementationTable) {
122 indexT oldIdx = srcModule->interfaceImplementationTable.getIndex(implPair.first);
123 auto newName = mangleName(modId, implPair.second->name);
124 indexT newIdx = finalModule->interfaceImplementationTable.put_create(newName, implPair.second);
125 // printf("remapping interface %lld %lld to %lld %lld\n", implPair.second->implInterfaceIndex.first, implPair.second->implInterfaceIndex.second, ENTRY_MODULE_ID_CONST, interfaceRemapping[implPair.second->implInterfaceIndex.first][implPair.second->implInterfaceIndex.second]);
126 finalModule->interfaceImplementationTable[newIdx]->implInterfaceIndex = {ENTRY_MODULE_ID_CONST, interfaceRemapping[implPair.second->implInterfaceIndex.first][implPair.second->implInterfaceIndex.second]};
127 finalModule->interfaceImplementationTable[newIdx]->name = newName;
128 interfaceImplRemapping[modId][oldIdx] = newIdx;
129 }
130 }
131 }

134 for (const auto& modPair : compilerCtx->getCompiledModules()) {
135 indexT modId = modPair.first;
136 const auto& srcModule = modPair.second;
137 for (const auto& globalPair : srcModule->globalVariables) {

147 void IRLinker::linkFunctions() {
148 // Pass 1: Create declarations for all functions to handle forward calls
149 for (const auto& modPair : compilerCtx->getCompiledModules()) {
150 indexT modId = modPair.first;
151 const auto& srcModule = modPair.second;
152 for (const auto& funcPair : srcModule->functionTable) {
153 indexT oldIdx = srcModule->functionTable.getIndex(funcPair.first);
154 auto newName = mangleName(modId, funcPair.second->name);
155
156 // Create a shell definition. The body will be filled in Pass 2.
157 auto newFuncDef = std::make_shared<IRFunctionDefinition>(*funcPair.second);
158 newFuncDef->name = newName;
159 newFuncDef->codeBlock.clear();
160
161 for (auto &var : newFuncDef->variableTable.getVariables()) {
162 *var = *patchType(var);
163 }
164 for (auto &param : newFuncDef->argumentTypes) {
165 *param = *patchType(param);
166 }
167
168 indexT newIdx = finalModule->functionTable.put_create(newName, newFuncDef);
169 functionRemapping[modId][oldIdx] = newIdx;
170
171 if (funcPair.first == L"yoimiya_glob_initializer") {
172 globInitializerIndexes.emplace_back(newIdx);
173 }
174 }
175 }
176
177 // Pass 2: Link bodies and patch instructions
178 for (const auto& modPair : compilerCtx->getCompiledModules()) {
179 indexT modId = modPair.first;
180 const auto& srcModule = modPair.second;
181 for (const auto& funcPair : srcModule->functionTable) {
182 indexT oldIdx = srcModule->functionTable.getIndex(funcPair.first);
183 indexT newIdx = functionRemapping.at(modId).at(oldIdx);
184
185 const auto& srcFunc = funcPair.second;
186 auto& destFunc = finalModule->functionTable[newIdx];
187
188 // Copy variable table, local vars don't need remapping
189 destFunc->returnType = patchType(srcFunc->returnType);
190 destFunc->variableTable = srcFunc->variableTable;
191
192 // Copy and patch code blocks
193 for (const auto& srcBlock : srcFunc->codeBlock) {
194 auto newBlock = std::make_shared<IRCodeBlock>();
195 for (const auto& srcInstr : srcBlock->getIRArray()) {
196 newBlock->insert(patchInstruction(srcInstr, modId));
197 }
198 destFunc->codeBlock.push_back(newBlock);
199 }
200 }
201 }

204 IR IRLinker::patchInstruction(const IR& instr, indexT currentModuleId) {
205 IR newInstr = instr;
206 switch (instr.opcode) {
207 case IR::Opcode::push_string: {
208 auto moduleId = instr.operands[0].value.symbolIndex;
209 auto stringIndex = instr.operands[1].value.symbolIndex;
210 auto newStringIndex = stringRemapping.at(moduleId).at(stringIndex);
211 newInstr.operands[1].value.symbolIndex = newStringIndex;
212 break;
213 }
214 case IR::Opcode::invoke:
215 case IR::Opcode::invoke_virtual:
216 case IR::Opcode::invoke_dangling:
217 case IR::Opcode::load_global:
218 case IR::Opcode::new_struct:
219 case IR::Opcode::new_array_struct:
220 case IR::Opcode::new_array_interface:
221 case IR::Opcode::new_dynamic_array_struct:
222 case IR::Opcode::new_dynamic_array_interface:
223 case IR::Opcode::construct_interface_impl:
224 case IR::Opcode::typeid_struct:
225 case IR::Opcode::typeid_interface:
226 case IR::Opcode::dyn_cast_struct: {
227 auto moduleId = instr.operands[0].value.symbolIndex;
228 auto symbolIndex = instr.operands[1].value.symbolIndex;
229 newInstr.operands[0].value.symbolIndex = ENTRY_MODULE_ID_CONST;
230 switch (instr.opcode) {
231 case IR::Opcode::invoke_dangling:
232 case IR::Opcode::invoke:
233 newInstr.operands[1].value.symbolIndex = functionRemapping.at(moduleId).at(symbolIndex);
234 break;
235 case IR::Opcode::load_global:
236 newInstr.operands[1].value.symbolIndex = globalRemapping.at(moduleId).at(symbolIndex);
237 break;
238 case IR::Opcode::new_struct:
239 case IR::Opcode::new_array_struct:
240 case IR::Opcode::new_dynamic_array_struct:
241 case IR::Opcode::dyn_cast_struct:
242 newInstr.operands[1].value.symbolIndex = structRemapping.at(moduleId).at(symbolIndex);
243 break;
244 case IR::Opcode::invoke_virtual:
245 case IR::Opcode::new_array_interface:
246 case IR::Opcode::new_dynamic_array_interface:
247 newInstr.operands[1].value.symbolIndex = interfaceRemapping.at(moduleId).at(symbolIndex);
248 break;
249 case IR::Opcode::construct_interface_impl:
250 newInstr.operands[1].value.symbolIndex = interfaceImplRemapping.at(moduleId).at(symbolIndex);
251 break;
252 default: break;
253 }
254 break;
255 }
256 case IR::Opcode::typeid_object_non_stack:
257 case IR::Opcode::dyn_cast_any: {
258 auto valueType = static_cast<IRValueType::valueType>(instr.operands[0].value.symbolIndex);
259 switch (valueType) {
260 case IRValueType::valueType::structObject:
261 newInstr.operands[1].value.symbolIndex = ENTRY_MODULE_ID_CONST;
262 newInstr.operands[2].value.symbolIndex = structRemapping.at(instr.operands[1].value.symbolIndex).at(instr.operands[2].value.symbolIndex);
263 break;
264 case IRValueType::valueType::interfaceObject:
265 newInstr.operands[1].value.symbolIndex = ENTRY_MODULE_ID_CONST;
266 newInstr.operands[2].value.symbolIndex = interfaceRemapping.at(instr.operands[1].value.symbolIndex).at(instr.operands[2].value.symbolIndex);
267 break;
268 default:
269 break;
270 }
271 break;
272 }
273 default:
274 break;
275 }
276
277 return newInstr;
278 }

280 std::shared_ptr<IRValueType> IRLinker::patchType(const std::shared_ptr<IRValueType> &oldType) {
281 // map the old type to the new type
282 if (oldType->typeAffiliateModule == ENTRY_MODULE_ID_CONST) {
283 return oldType;
284 }
285 std::shared_ptr<IRValueType> newType = managedPtr(*oldType);
286 switch (oldType->type) {
287 case IRValueType::valueType::structObject: {
288 auto newIndex = structRemapping[oldType->typeAffiliateModule][oldType->typeIndex];
289 newType->typeAffiliateModule = ENTRY_MODULE_ID_CONST;
290 newType->typeIndex = newIndex;
291 break;
292 }
293 case IRValueType::valueType::interfaceObject: {
294 auto newIndex = interfaceRemapping[oldType->typeAffiliateModule][oldType->typeIndex];
295 newType->typeAffiliateModule = ENTRY_MODULE_ID_CONST;
296 newType->typeIndex = newIndex;
297 break;
298 }
299 case IRValueType::valueType::virtualMethod: {
300 auto newIndex = functionRemapping[oldType->typeAffiliateModule][oldType->typeIndex];
301 newType->typeAffiliateModule = ENTRY_MODULE_ID_CONST;
302 newType->typeIndex = newIndex;
303 break;
304 }
305 default: {
306 // Other types don't need patching
307 break;
308 }
309 }
310 return newType;

312 void IRLinker::createEntryFunction() {
313 IRFunctionDefinition::Builder entryBuilder;
314 auto entry = entryBuilder.setName(L"yoimiya_entry").setReturnType(compilerCtx->getIntObjectType()).setDebugInfo({L"<entry>", 0, 0}).yield();
315 auto entryIndex = this->finalModule->functionTable.put_create(L"yoimiya_entry", entry);
316 IRBuilder builder(compilerCtx, finalModule, entry);
317 builder.switchCodeBlock(builder.createCodeBlock());
318 for (auto &initIdx: globInitializerIndexes) {
319 if (finalModule->functionTable[initIdx]->hasAttribute(IRFunctionDefinition::FunctionAttrs::Unreachable))
320 continue;
321 builder.invokeOp(initIdx, 0, compilerCtx->getIntObjectType());
322 }
323 if (compilerCtx->getBuildConfig()->buildType == IRBuildConfig::BuildType::executable) {
324 try {
325 builder.invokeOp(finalModule->functionTable.getIndex(L"yoi_main"), 0, compilerCtx->getIntObjectType());
326 builder.retOp();
327 } catch (const std::out_of_range&) {
328 panic(0, 0, "Entry function not found for executable build!");
329 return;
330 }
331 } else {
332 builder.pushOp(IR::Opcode::push_integer, IROperand{IROperand::operandType::integer, IROperand::operandValue{(int64_t)0}});
333 builder.retOp();
334 }
335 builder.yield();
336 }

345 void IRLinker::patchIRFFITable() {
346 for (auto &funcPair : compilerCtx->getIRFFITable()->exportedFunctionTable) {
347 funcPair.second = {ENTRY_MODULE_ID_CONST, functionRemapping.at(std::get<0>(funcPair.second)).at(std::get<1>(funcPair.second)), std::get<2>(funcPair.second)};
348 }
349 for (auto &libPair : compilerCtx->getIRFFITable()->importedLibraries) {
350 for (auto &funcPair : libPair.second.importedFunctionTable) {
351 funcPair.second->returnType = patchType(funcPair.second->returnType);
352 for (auto &param : funcPair.second->argumentTypes) {
353 *param = *patchType(param);
354 }
355 for (auto &ret : funcPair.second->variableTable.getVariables()) {
356 *ret = *patchType(ret);
357 }
358 }
359 }
360 for (auto &foreignTypePair : compilerCtx->getIRFFITable()->foreignTypeTable) {
361 foreignTypePair.second = patchType(foreignTypePair.second);
362 }

365 std::tuple<IRValueType::valueType, indexT, indexT>
366 IRLinker::patchUniqueKey(const std::tuple<IRValueType::valueType, indexT, indexT> &key) {
367 switch (std::get<0>(key)) {
368 case IRValueType::valueType::structObject: {
369 auto newIndex = structRemapping[std::get<1>(key)][std::get<2>(key)];
370 return std::make_tuple(IRValueType::valueType::structObject, ENTRY_MODULE_ID_CONST, newIndex);
371 }
372 case IRValueType::valueType::interfaceObject: {
373 auto newIndex = interfaceRemapping[std::get<1>(key)][std::get<2>(key)];
374 return std::make_tuple(IRValueType::valueType::interfaceObject, ENTRY_MODULE_ID_CONST, newIndex);
375 }
376 case IRValueType::valueType::virtualMethod: {
377 auto newIndex = functionRemapping[std::get<1>(key)][std::get<2>(key)];
378 return std::make_tuple(IRValueType::valueType::virtualMethod, ENTRY_MODULE_ID_CONST, newIndex);
379 }
380 default: {
381 // Other types don't need patching
382 return key;
383 }
384 }
