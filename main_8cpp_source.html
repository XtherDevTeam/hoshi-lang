22std::string getOutputExtension(yoi::IRBuildConfig::BuildType type,
23 std::wstring platform) {
24 if (type == yoi::IRBuildConfig::BuildType::executable) {
25 if (platform == L"windows") return ".exe";

27 } else if (type == yoi::IRBuildConfig::BuildType::library) {
