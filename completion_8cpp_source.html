12// Look up a function's return type from the visitor's IR module.
13static std::string getReturnTypeFromIR(Document *doc, const std::string &funcName) {
14 if (!doc || !doc->irModule) return "";
15 yoi::wstr wName = yoi::string2wstring(funcName);
16 // Check function overloads first
17 if (doc->irModule->functionOverloadIndexies.contains(wName)) {
18 auto &overloads = doc->irModule->functionOverloadIndexies[wName];
19 if (!overloads.empty()) {
20 auto idx = overloads[0];
21 auto &funcDef = *doc->irModule->functionTable[idx];
22 // funcDef has returnType info; parse from the name string
23 std::string detail = yoi::wstring2string(funcDef.name);
24 size_t colon = detail.rfind(':');
25 if (colon != std::string::npos) {
26 std::string retType = detail.substr(colon + 1);
27 while (!retType.empty() && std::isspace(retType.front())) retType.erase(0, 1);
28 size_t tmpl = retType.find('<');
29 if (tmpl != std::string::npos) retType = retType.substr(0, tmpl);
30 size_t dot = retType.rfind('.');
31 if (dot != std::string::npos) retType = retType.substr(dot + 1);
32 return retType;
33 }
34 }
35 }
36 return "";
37}

39CompletionList CompletionProvider::provide(Document *doc, const Position &pos) {
40 CompletionList result;
41 result.isIncomplete = false;
42
43 if (!doc) return result;
44
45 int charPos = static_cast<int>(pos.character);
46 int lineNum = static_cast<int>(pos.line);
47
48 // Find the line text
49 std::string lineText;
50 {
51 int lineIdx = 0;
52 size_t start = 0;
53 std::string text = yoi::wstring2string(doc->text);
54 for (size_t i = 0; i < text.size(); i++) {
55 if (text[i] == '\n') {
56 if (lineIdx == lineNum) {
57 lineText = text.substr(start, i - start);
58 break;
59 }
60 lineIdx++;
61 start = i + 1;
62 }
63 }
64 if (lineText.empty() && lineIdx == lineNum) {
65 lineText = text.substr(start);
66 }
67 }
68
69 // Get the prefix (word before cursor)
70 std::string prefix;
71 if (charPos > 0 && charPos <= static_cast<int>(lineText.size())) {
72 std::string before = lineText.substr(0, charPos);
73 size_t wordStart = before.size();
74 while (wordStart > 0 && (std::isalnum(before[wordStart - 1]) || before[wordStart - 1] == '_'))
75 wordStart--;
76 prefix = before.substr(wordStart);
77 }
78
79 // Detect dot-access: cursor after a dot or after an identifier preceded by a dot
80 bool afterDot = false;
81 std::string dotPrefix; // The identifier before the dot (module alias / struct name)
82 if (charPos > 0 && charPos <= static_cast<int>(lineText.size())) {
83 std::string before = lineText.substr(0, charPos);
84 while (!before.empty() && std::isspace(before.back()))
85 before.pop_back();
86 // Check if the character immediately before the cursor is a dot,
87 // or if there's a dot before the current word
88 if (!before.empty()) {
89 // Find the last dot position
90 size_t lastDot = before.rfind('.');
91 if (lastDot != std::string::npos) {
92 // Check if cursor is right after the dot (no word yet)
93 if (lastDot == before.size() - 1) {
94 afterDot = true;
95 // Extract the identifier before the dot
96 std::string beforeDot = before.substr(0, lastDot);
97 // Get the last identifier before the dot
98 size_t idEnd = beforeDot.size();
99 size_t idStart = idEnd;
100 while (idStart > 0 && (std::isalnum(beforeDot[idStart - 1]) || beforeDot[idStart - 1] == '_'))
101 idStart--;
102 if (idStart < idEnd) {
103 dotPrefix = beforeDot.substr(idStart, idEnd - idStart);
104 }
105 } else {
106 // There's a word after the dot — check if it's part of the current identifier
107 std::string afterDotStr = before.substr(lastDot + 1);
108 // If prefix starts at or after lastDot+1, we're doing member completion
109 if (prefix.rfind(afterDotStr, 0) == 0 || afterDotStr.rfind(prefix, 0) == 0) {
110 afterDot = true;
111 prefix = afterDotStr; // Use the part after dot as prefix
112 std::string beforeDot = before.substr(0, lastDot);
113 size_t idEnd = beforeDot.size();
114 size_t idStart = idEnd;
115 while (idStart > 0 && (std::isalnum(beforeDot[idStart - 1]) || beforeDot[idStart - 1] == '_'))
116 idStart--;
117 if (idStart < idEnd) {
118 dotPrefix = beforeDot.substr(idStart, idEnd - idStart);
119 }
120 }
121 }
122 }
123 }
124 }
125
126 // Gather completions
127 if (!doc->parseSucceeded && afterDot && !dotPrefix.empty()) {
128 // Parse failed but we have a dot-prefix — try to resolve the module
129 // from the raw text. This handles incomplete code like "str." where
130 // the parser throws on the trailing dot.
131 result.items = resolveModuleCompletionsFromText(doc, dotPrefix, prefix);
132 // Fall back to keywords if module resolution failed
133 if (result.items.empty()) {
134 result.items = keywordCompletions(prefix);
135 }
136 return result;
137 } else if (!doc->parseSucceeded) {
138 // Parse failed — return keywords as a fallback
139 result.items = keywordCompletions(prefix);
140 return result;
141 } else if (afterDot && !dotPrefix.empty()) {
142 // Member access — try to match dotPrefix to a module alias or struct
143 result.items = memberCompletions(doc, dotPrefix, prefix);
144 // If member completions returned nothing, fall back to global completions
145 if (result.items.empty()) {
146 result.items = symbolCompletions(doc, prefix);
147 auto crossItems = crossModuleCompletions(doc, prefix);
148 result.items.insert(result.items.end(), crossItems.begin(), crossItems.end());
149 auto kwCompletions = keywordCompletions(prefix);
150 result.items.insert(result.items.end(), kwCompletions.begin(), kwCompletions.end());
151 }
152 } else if (afterDot) {
153 // Dot without a clear prefix (e.g., at start of line, or after a number)
154 // Fall through to global completions instead of returning empty
155 result.items = symbolCompletions(doc, prefix);
156 auto crossItems = crossModuleCompletions(doc, prefix);
157 result.items.insert(result.items.end(), crossItems.begin(), crossItems.end());
158 auto kwCompletions = keywordCompletions(prefix);
159 result.items.insert(result.items.end(), kwCompletions.begin(), kwCompletions.end());
160 } else {
161 // Symbol + keyword completions (global scope)
162 result.items = symbolCompletions(doc, prefix);
163 auto crossItems = crossModuleCompletions(doc, prefix);
164 result.items.insert(result.items.end(), crossItems.begin(), crossItems.end());
165 auto kwCompletions = keywordCompletions(prefix);
166 result.items.insert(result.items.end(), kwCompletions.begin(), kwCompletions.end());
167 }
168
169 // Sort alphabetically
170 std::sort(result.items.begin(), result.items.end(),
171 [](const CompletionItem &a, const CompletionItem &b) {
172 return a.sortText.value_or(a.label) < b.sortText.value_or(b.label);
173 });
174
175 return result;
176}

178std::vector<CompletionItem> CompletionProvider::memberCompletions(Document *doc, const std::string &parent, const std::string &prefix) {
179 std::vector<CompletionItem> result;
180 yoi::wstr wParent = yoi::string2wstring(parent);
181
182 // Check document-local symbols: find a struct/interface or variable with name == parent
183 for (auto &sym : doc->symbols) {
184 if (sym.name == wParent) {
185 // Case 1: struct/interface — offer its children
186 if (!sym.children.empty()) {
187 for (auto &child : sym.children) {
188 std::string name = yoi::wstring2string(child.name);
189 if (name.rfind(prefix, 0) != 0) continue;
190 CompletionItem item = symbolToCompletionItem(child);
191 item.label = name;
192 item.insertText = name;
193 result.push_back(item);
194 }
195 return result; // Found struct/interface with children
196 }
197 // Case 2: typed variable — look up its type and offer methods
198 if (sym.kind == HoshiSymbolKind::Variable && !sym.typeInfo.empty()) {
199 result = completionsForType(doc, sym.typeInfo, prefix);
200 if (!result.empty()) return result;
201 }
202 // If it's a ModuleAlias or unresolved, fall through to cross-module search
203 }
204 }
205
206 // Check cross-module symbols: find a ModuleAlias with name == parent
207 for (auto &sym : doc->symbols) {
208 if (sym.kind == HoshiSymbolKind::ModuleAlias && sym.name == wParent) {
209 // This is a module alias — offer all symbols from the resolved module
210 for (auto &cross : doc->crossModuleSymbols) {
211 std::string name = yoi::wstring2string(cross.name);
212 if (name.rfind(prefix, 0) != 0) continue;
213 CompletionItem item = symbolToCompletionItem(cross);
214 item.label = name;
215 item.insertText = name;
216 item.documentation = std::string("From module ") + yoi::wstring2string(sym.importPath);
217 result.push_back(item);
218 }
219 return result;
220 }
221 }
222
223 // Check cross-module symbols: find a struct/interface by name and offer children
224 for (auto &cross : doc->crossModuleSymbols) {
225 if (cross.name == wParent && !cross.children.empty() &&
226 (cross.kind == HoshiSymbolKind::Struct || cross.kind == HoshiSymbolKind::Interface)) {
227 for (auto &child : cross.children) {
228 std::string name = yoi::wstring2string(child.name);
229 if (name.rfind(prefix, 0) != 0) continue;
230 CompletionItem item = symbolToCompletionItem(child);
231 item.label = name;
232 item.insertText = name;
233 item.documentation = std::string("From ") + yoi::wstring2string(cross.sourceFile);
234 result.push_back(item);
235 }
236 if (!result.empty()) return result;
237 }
238 }
239
240 // Also check cross-module symbols directly by parentName (for methods/fields)
241 for (auto &cross : doc->crossModuleSymbols) {
242 if (cross.parentName == wParent) {
243 std::string name = yoi::wstring2string(cross.name);
244 if (name.rfind(prefix, 0) != 0) continue;
245 CompletionItem item = symbolToCompletionItem(cross);
246 item.label = name;
247 item.insertText = name;
248 result.push_back(item);
249 }
250 }
251
252 return result;
253}

277CompletionItem CompletionProvider::symbolToCompletionItem(const Symbol &sym) {
278 CompletionItem item;
279 item.label = yoi::wstring2string(sym.name);
280 item.insertText = yoi::wstring2string(sym.name);
281 item.sortText = yoi::wstring2string(sym.name);
282 item.detail = yoi::wstring2string(sym.detail);
283
284 switch (sym.kind) {
285 case HoshiSymbolKind::Function: item.kind = CompletionItemKind::Function; break;
286 case HoshiSymbolKind::Struct: item.kind = CompletionItemKind::Struct; break;
287 case HoshiSymbolKind::Interface: item.kind = CompletionItemKind::Interface; break;
288 case HoshiSymbolKind::Variable: item.kind = CompletionItemKind::Variable; break;
289 case HoshiSymbolKind::TypeAlias: item.kind = CompletionItemKind::TypeParameter; break;
290 case HoshiSymbolKind::Enum: item.kind = CompletionItemKind::Enum; break;
291 case HoshiSymbolKind::Module:
292 case HoshiSymbolKind::ModuleAlias: item.kind = CompletionItemKind::Module; break;
293 case HoshiSymbolKind::Method: item.kind = CompletionItemKind::Method; break;
294 case HoshiSymbolKind::Field: item.kind = CompletionItemKind::Field; break;
295 case HoshiSymbolKind::Constructor: item.kind = CompletionItemKind::Constructor; break;
296 case HoshiSymbolKind::Finalizer: item.kind = CompletionItemKind::Method; break;
297 case HoshiSymbolKind::EnumMember: item.kind = CompletionItemKind::EnumMember; break;
298 case HoshiSymbolKind::Import:
299 case HoshiSymbolKind::Export: item.kind = CompletionItemKind::Reference; break;
300 case HoshiSymbolKind::Concept_: item.kind = CompletionItemKind::Interface; break;
301 default: item.kind = CompletionItemKind::Text; break;
302 }
303
304 if (!sym.typeInfo.empty())
305 item.documentation = "Type: " + yoi::wstring2string(sym.typeInfo);
306
307 return item;
308}

394std::vector<CompletionItem> CompletionProvider::resolveModuleCompletionsFromText(
395 Document *doc, const std::string &parent, const std::string &prefix) {
396 std::vector<CompletionItem> result;
397 if (!doc || !doc->projectIndex) return result;
398
399 // Search the raw text for "use <parent> \"...\"" or "import ... from \"...\"" patterns
400 std::string text = yoi::wstring2string(doc->text);
401 std::string importPath;
402
403 // Pattern 1: use <parent> "<path>"
404 std::string usePattern = "use " + parent + " \"";
405 size_t pos = text.find(usePattern);
406 if (pos != std::string::npos) {
407 size_t pathStart = pos + usePattern.size();
408 size_t pathEnd = text.find('"', pathStart);
409 if (pathEnd != std::string::npos) {
410 importPath = text.substr(pathStart, pathEnd - pathStart);
411 }
412 }
413
414 // Pattern 2: import ... from "<path>" (any import whose path might contain the module)
415 if (importPath.empty()) {
416 std::string importKw = "import ";
417 size_t ipos = 0;
418 while ((ipos = text.find(importKw, ipos)) != std::string::npos) {
419 size_t fromPos = text.find("from \"", ipos);
420 if (fromPos != std::string::npos) {
421 size_t pathStart = fromPos + 6; // strlen("from \"")
422 size_t pathEnd = text.find('"', pathStart);
423 if (pathEnd != std::string::npos) {
424 std::string maybePath = text.substr(pathStart, pathEnd - pathStart);
425 // Check if the module path ends with the parent name
426 if (maybePath == parent ||
427 maybePath.rfind("/" + parent) == maybePath.size() - parent.size() - 1 ||
428 maybePath.rfind("\\" + parent) == maybePath.size() - parent.size() - 1) {
429 importPath = maybePath;
430 break;
431 }
432 }
433 }
434 ipos = fromPos != std::string::npos ? fromPos + 1 : ipos + 1;
435 }
436 }
437
438 if (importPath.empty()) {
439 // Not a module alias — try to find the variable's type from its declaration.
440 // Look for "let <parent> = <Module>.<Type>(...)" or "let <parent>: <Type>"
441 std::string letPattern = "let " + parent + " = ";
442 size_t letPos = text.find(letPattern);
443 if (letPos == std::string::npos) {
444 letPattern = "let " + parent + ": ";
445 letPos = text.find(letPattern);
446 }
447 if (letPos != std::string::npos) {
448 std::string afterLet = text.substr(letPos + letPattern.size());
449 // Extract the type: for "str.Str(" → module="str", type="Str"
450 size_t dotPos = afterLet.find('.');
451 std::string typeName, moduleName;
452 if (dotPos != std::string::npos && dotPos < afterLet.find('(')) {
453 // Module.Type pattern — extract module and type
454 moduleName = afterLet.substr(0, dotPos);
455 size_t typeStart = dotPos + 1;
456 size_t typeEnd = typeStart;
457 while (typeEnd < afterLet.size() && (std::isalnum(afterLet[typeEnd]) || afterLet[typeEnd] == '_'))
458 typeEnd++;
459 typeName = afterLet.substr(typeStart, typeEnd - typeStart);
460 } else {
461 size_t typeEnd = 0;
462 while (typeEnd < afterLet.size() && (std::isalnum(afterLet[typeEnd]) || afterLet[typeEnd] == '_'))
463 typeEnd++;
464 typeName = afterLet.substr(0, typeEnd);
465 }
466 if (!typeName.empty() && doc->projectIndex) {
467 // Index the module and resolve the type
468 if (!moduleName.empty()) {
469 std::string fp = doc->uri;
470 if (fp.rfind("file://", 0) == 0) fp = fp.substr(7);
471 std::string useP = "use " + moduleName + " \"";
472 size_t upos = text.find(useP);
473 if (upos != std::string::npos) {
474 size_t ps = upos + useP.size();
475 size_t pe = text.find('"', ps);
476 if (pe != std::string::npos) {
477 std::string resolvedPath = doc->projectIndex->resolveModule(
478 text.substr(ps, pe - ps), fp);
479 if (!resolvedPath.empty()) {
480 doc->projectIndex->indexAndGet(resolvedPath);
481 }
482 }
483 }
484 }
485
486 // If lowercase (function call), look up the function's return type
487 std::string lookupType = typeName;
488 if (std::islower(typeName[0]) && !moduleName.empty()) {
489 // First try the visitor's IR module for precise return type
490 std::string irRetType = getReturnTypeFromIR(doc, typeName);
491 if (!irRetType.empty()) {
492 lookupType = irRetType;
493 } else {
494 // Fall back to symbol-based lookup
495 yoi::wstr wFunc = yoi::string2wstring(typeName);
496 yoi::vec<Symbol> allSyms;
497 doc->projectIndex->getAllSymbols(allSyms);
498 for (auto &sym : allSyms) {
499 if (sym.name == wFunc && sym.kind == HoshiSymbolKind::Function && !sym.typeInfo.empty()) {
500 std::string retType = yoi::wstring2string(sym.typeInfo);
501 size_t lastDot = retType.rfind('.');
502 if (lastDot != std::string::npos) retType = retType.substr(lastDot + 1);
503 size_t tmpl = retType.find('<');
504 if (tmpl != std::string::npos) retType = retType.substr(0, tmpl);
505 while (!retType.empty() && std::isspace(retType.back())) retType.pop_back();
506 if (!retType.empty()) { lookupType = retType; break; }
507 }
508 }
509 }
510 }
511
512 // Look up the type name for member completion
513 if (std::isupper(lookupType[0])) {
514 auto completions = completionsForTypeNameFromModules(doc, lookupType, prefix);
515 if (!completions.empty()) return completions;
516 }
517 }
518 }
519 return result;
520 }
521
522 // Resolve the module path relative to the document's directory
523 std::string filePath = doc->uri;
524 if (filePath.rfind("file://", 0) == 0) {
525 filePath = filePath.substr(7);
526 }
527 std::string resolvedPath = doc->projectIndex->resolveModule(importPath, filePath);
528 if (resolvedPath.empty()) return result;
529
530 // Get symbols from the resolved module
531 const auto &modSymbols = doc->projectIndex->indexAndGet(resolvedPath);
532
533 // Build completion items
534 for (const auto &sym : modSymbols) {
535 std::string name = yoi::wstring2string(sym.name);
536 if (!prefix.empty() && name.rfind(prefix, 0) != 0) continue;
537
538 CompletionItem item = symbolToCompletionItem(sym);
539 item.label = name;
540 item.insertText = name;
541 item.sortText = name;
542 item.documentation = std::string("From ") + resolvedPath;
543 result.push_back(item);
544 }
545
546 return result;
547}

549std::vector<CompletionItem> CompletionProvider::completionsForType(
550 Document *doc, const yoi::wstr &typeName, const std::string &prefix) {
551 std::vector<CompletionItem> result;
552 bool found = false;
553
554 // Search local symbols for a struct/interface with this name
555 for (auto &sym : doc->symbols) {
556 if (sym.name == typeName &&
557 (sym.kind == HoshiSymbolKind::Struct || sym.kind == HoshiSymbolKind::Interface)) {
558 found = true;
559 // Include children (declared inside the struct)
560 for (auto &child : sym.children) {
561 std::string name = yoi::wstring2string(child.name);
562 if (name.rfind(prefix, 0) != 0) continue;
563 CompletionItem item = symbolToCompletionItem(child);
564 item.label = name;
565 item.insertText = name;
566 item.documentation = std::string("From ") + yoi::wstring2string(sym.name);
567 result.push_back(item);
568 }
569 }
570 // Also include symbols whose parentName matches (methods from impl blocks)
571 if (sym.parentName == typeName &&
572 (sym.kind == HoshiSymbolKind::Method || sym.kind == HoshiSymbolKind::Field ||
573 sym.kind == HoshiSymbolKind::Constructor || sym.kind == HoshiSymbolKind::Finalizer)) {
574 std::string name = yoi::wstring2string(sym.name);
575 if (name.rfind(prefix, 0) != 0) continue;
576 CompletionItem item = symbolToCompletionItem(sym);
577 item.label = name;
578 item.insertText = name;
579 item.documentation = std::string("From ") + yoi::wstring2string(typeName);
580 result.push_back(item);
581 }
582 }
583 if (found && !result.empty()) return result;
584
585 // Search cross-module symbols
586 yoi::vec<Symbol> allCross;
587 for (auto &cross : doc->crossModuleSymbols) allCross.push_back(cross);
588 // Also get all from ProjectIndex
589 if (doc->projectIndex) {
590 doc->projectIndex->getAllSymbols(allCross);
591 }
592
593 for (auto &cross : allCross) {
594 if (cross.name == typeName &&
595 (cross.kind == HoshiSymbolKind::Struct || cross.kind == HoshiSymbolKind::Interface)) {
596 found = true;
597 for (auto &child : cross.children) {
598 std::string name = yoi::wstring2string(child.name);
599 if (name.rfind(prefix, 0) != 0) continue;
600 CompletionItem item = symbolToCompletionItem(child);
601 item.label = name;
602 item.insertText = name;
603 item.documentation = std::string("From ") + yoi::wstring2string(typeName);
604 result.push_back(item);
605 }
606 }
607 if (cross.parentName == typeName &&
608 (cross.kind == HoshiSymbolKind::Method || cross.kind == HoshiSymbolKind::Field ||
609 cross.kind == HoshiSymbolKind::Constructor || cross.kind == HoshiSymbolKind::Finalizer)) {
610 std::string name = yoi::wstring2string(cross.name);
611 if (name.rfind(prefix, 0) != 0) continue;
612 CompletionItem item = symbolToCompletionItem(cross);
613 item.label = name;
614 item.insertText = name;
615 item.documentation = std::string("From ") + yoi::wstring2string(typeName);
616 result.push_back(item);
