52int clock_gettime(int clk_id, struct timespec *tp) {
53 if (tp == NULL)
54 return -1;
55
56 if (clk_id == CLOCK_REALTIME) {
57 FILETIME ft;
58 GetSystemTimePreciseAsFileTime(&ft);
59
60 uint64_t t = ((uint64_t)ft.dwHighDateTime << 32) | ft.dwLowDateTime;
61
62 t -= DELTA_EPOCH_IN_MICROSECS;
63
64 tp->tv_sec = (time_t)(t / 10000000); // 100ns to seconds
65 tp->tv_nsec = (long)((t % 10000000) * 100); // Remaining 100ns to ns
66
67 return 0;
68 } else if (clk_id == CLOCK_MONOTONIC) {
69 static LARGE_INTEGER freq = {0};
70 LARGE_INTEGER count;
71
72 // Get frequency once (it doesn't change after boot)
73 if (freq.QuadPart == 0) {
74 if (!QueryPerformanceFrequency(&freq))
75 return -1;
76 }
77
78 if (!QueryPerformanceCounter(&count))
79 return -1;
80
81 // Convert QPC units to seconds and nanoseconds
82 tp->tv_sec = (time_t)(count.QuadPart / freq.QuadPart);
83
84 // Calculate remainder carefully to avoid overflow before division
85 // logic: (count % freq) * 1e9 / freq
86 tp->tv_nsec = (long)(((count.QuadPart % freq.QuadPart) * 1000000000) / freq.QuadPart);
87
88 return 0;
89 }

91 return -1; // Unknown clock ID
92}
