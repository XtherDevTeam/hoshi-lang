24typedef bool (heap_page_visitor_fun)(mi_heap_t* heap, mi_page_queue_t* pq, mi_page_t* page, void* arg1, void* arg2);

26// Visit all pages in a heap; returns `false` if break was called.
27static bool mi_heap_visit_pages(mi_heap_t* heap, heap_page_visitor_fun* fn, void* arg1, void* arg2)
28{
29 if (heap==NULL || heap->page_count==0) return 0;
30
31 // visit all pages
32 #if MI_DEBUG>1
33 size_t total = heap->page_count;
34 size_t count = 0;
35 #endif
36
37 for (size_t i = 0; i <= MI_BIN_FULL; i++) {
38 mi_page_queue_t* pq = &heap->pages[i];
39 mi_page_t* page = pq->first;
40 while(page != NULL) {
41 mi_page_t* next = page->next; // save next in case the page gets removed from the queue
42 mi_assert_internal(mi_page_heap(page) == heap);
43 #if MI_DEBUG>1
44 count++;
45 #endif
46 if (!fn(heap, pq, page, arg1, arg2)) return false;
47 page = next; // and continue
48 }
49 }
50 mi_assert_internal(count == total);
51 return true;
52}

78/* -----------------------------------------------------------
79 "Collect" pages by migrating `local_free` and `thread_free`
80 lists and freeing empty pages. This is done when a thread
81 stops (and in that case abandons pages if there are still
82 blocks alive)
83----------------------------------------------------------- */
84
85typedef enum mi_collect_e {
86 MI_NORMAL,
87 MI_FORCE,
88 MI_ABANDON
89} mi_collect_t;
90
91
92static bool mi_heap_page_collect(mi_heap_t* heap, mi_page_queue_t* pq, mi_page_t* page, void* arg_collect, void* arg2 ) {
93 MI_UNUSED(arg2);
94 MI_UNUSED(heap);
95 mi_assert_internal(mi_heap_page_is_valid(heap, pq, page, NULL, NULL));
96 mi_collect_t collect = *((mi_collect_t*)arg_collect);
97 _mi_page_free_collect(page, collect >= MI_FORCE);
98 if (collect == MI_FORCE) {
99 // note: call before a potential `_mi_page_free` as the segment may be freed if this was the last used page in that segment.
100 mi_segment_t* segment = _mi_page_segment(page);
101 _mi_segment_collect(segment, true /* force? */);
102 }
103 if (mi_page_all_free(page)) {
104 // no more used blocks, free the page.
105 // note: this will free retired pages as well.
106 _mi_page_free(page, pq, collect >= MI_FORCE);
107 }
108 else if (collect == MI_ABANDON) {
109 // still used blocks but the thread is done; abandon the page
110 _mi_page_abandon(page, pq);
111 }
112 return true; // don't break
113}

124static void mi_heap_collect_ex(mi_heap_t* heap, mi_collect_t collect)
125{
126 if (heap==NULL || !mi_heap_is_initialized(heap)) return;
127
128 const bool force = (collect >= MI_FORCE);
129 _mi_deferred_free(heap, force);
130
131 // python/cpython#112532: we may be called from a thread that is not the owner of the heap
132 const bool is_main_thread = (_mi_is_main_thread() && heap->thread_id == _mi_thread_id());
133
134 // note: never reclaim on collect but leave it to threads that need storage to reclaim
135 const bool force_main =
136 #ifdef NDEBUG
137 collect == MI_FORCE
138 #else
139 collect >= MI_FORCE
140 #endif
141 && is_main_thread && mi_heap_is_backing(heap) && !heap->no_reclaim;
142
143 if (force_main) {
144 // the main thread is abandoned (end-of-program), try to reclaim all abandoned segments.
145 // if all memory is freed by now, all segments should be freed.
146 // note: this only collects in the current subprocess
147 _mi_abandoned_reclaim_all(heap, &heap->tld->segments);
148 }
149
150 // if abandoning, mark all pages to no longer add to delayed_free
151 if (collect == MI_ABANDON) {
152 mi_heap_visit_pages(heap, &mi_heap_page_never_delayed_free, NULL, NULL);
153 }
154
155 // free all current thread delayed blocks.
156 // (if abandoning, after this there are no more thread-delayed references into the pages.)
157 _mi_heap_delayed_free_all(heap);
158
159 // collect retired pages
160 _mi_heap_collect_retired(heap, force);
161
162 // collect all pages owned by this thread
163 mi_heap_visit_pages(heap, &mi_heap_page_collect, &collect, NULL);
164 mi_assert_internal( collect != MI_ABANDON || mi_atomic_load_ptr_acquire(mi_block_t,&heap->thread_delayed_free) == NULL );
165
166 // collect abandoned segments (in particular, purge expired parts of segments in the abandoned segment list)
167 // note: forced purge can be quite expensive if many threads are created/destroyed so we do not force on abandonment
168 _mi_abandoned_collect(heap, collect == MI_FORCE /* force? */, &heap->tld->segments);
169
170 // if forced, collect thread data cache on program-exit (or shared library unload)
171 if (force && is_main_thread && mi_heap_is_backing(heap)) {
172 _mi_thread_data_collect(); // collect thread data cache
173 }
174
175 // collect arenas (this is program wide so don't force purges on abandonment of threads)
176 _mi_arenas_collect(collect == MI_FORCE /* force purge? */);
177
178 // merge statistics
179 if (collect <= MI_FORCE) { _mi_stats_merge_thread(heap->tld); }
180}

218void _mi_heap_init(mi_heap_t* heap, mi_tld_t* tld, mi_arena_id_t arena_id, bool noreclaim, uint8_t tag) {
219 _mi_memcpy_aligned(heap, &_mi_heap_empty, sizeof(mi_heap_t));
220 heap->tld = tld;
221 heap->thread_id = _mi_thread_id();
222 heap->arena_id = arena_id;
223 heap->no_reclaim = noreclaim;
224 heap->tag = tag;
225 if (heap == tld->heap_backing) {
226 _mi_random_init(&heap->random);
227 }
228 else {
229 _mi_random_split(&tld->heap_backing->random, &heap->random);
230 }
231 heap->cookie = _mi_heap_random_next(heap) | 1;
232 heap->keys[0] = _mi_heap_random_next(heap);
233 heap->keys[1] = _mi_heap_random_next(heap);
234 _mi_heap_guarded_init(heap);
235 // push on the thread local heaps list
236 heap->next = heap->tld->heaps;
237 heap->tld->heaps = heap;
238}

277// called from `mi_heap_destroy` and `mi_heap_delete` to free the internal heap resources.
278static void mi_heap_free(mi_heap_t* heap) {
279 mi_assert(heap != NULL);
280 mi_assert_internal(mi_heap_is_initialized(heap));
281 if (heap==NULL || !mi_heap_is_initialized(heap)) return;
282 if (mi_heap_is_backing(heap)) return; // dont free the backing heap
283
284 // reset default
285 if (mi_heap_is_default(heap)) {
286 _mi_heap_set_default_direct(heap->tld->heap_backing);
287 }
288
289 // remove ourselves from the thread local heaps list
290 // linear search but we expect the number of heaps to be relatively small
291 mi_heap_t* prev = NULL;
292 mi_heap_t* curr = heap->tld->heaps;
293 while (curr != heap && curr != NULL) {
294 prev = curr;
295 curr = curr->next;
296 }
297 mi_assert_internal(curr == heap);
298 if (curr == heap) {
299 if (prev != NULL) { prev->next = heap->next; }
300 else { heap->tld->heaps = heap->next; }
301 }
302 mi_assert_internal(heap->tld->heaps != NULL);
303
304 // and free the used memory
305 mi_free(heap);
306}

325static bool _mi_heap_page_destroy(mi_heap_t* heap, mi_page_queue_t* pq, mi_page_t* page, void* arg1, void* arg2) {
326 MI_UNUSED(arg1);
327 MI_UNUSED(arg2);
328 MI_UNUSED(heap);
329 MI_UNUSED(pq);
330
331 // ensure no more thread_delayed_free will be added
332 _mi_page_use_delayed_free(page, MI_NEVER_DELAYED_FREE, false);
333
334 // stats
335 const size_t bsize = mi_page_block_size(page);
336 if (bsize > MI_MEDIUM_OBJ_SIZE_MAX) {
337 //if (bsize <= MI_LARGE_OBJ_SIZE_MAX) {
338 // mi_heap_stat_decrease(heap, malloc_large, bsize);
339 //}
340 //else
341 {
342 mi_heap_stat_decrease(heap, malloc_huge, bsize);
343 }
344 }
345 #if (MI_STAT>0)
346 _mi_page_free_collect(page, false); // update used count
347 const size_t inuse = page->used;
348 if (bsize <= MI_LARGE_OBJ_SIZE_MAX) {
349 mi_heap_stat_decrease(heap, malloc_normal, bsize * inuse);
350 #if (MI_STAT>1)
351 mi_heap_stat_decrease(heap, malloc_bins[_mi_bin(bsize)], inuse);
352 #endif
353 }
354 // mi_heap_stat_decrease(heap, malloc_requested, bsize * inuse); // todo: off for aligned blocks...
355 #endif
356
358 mi_assert_internal(mi_page_thread_free(page) == NULL);
359 page->used = 0;
360
361 // and free the page
362 // mi_page_free(page,false);
363 page->next = NULL;
364 page->prev = NULL;
365 _mi_segment_page_free(page,false /* no force? */, &heap->tld->segments);
366
367 return true; // keep going
368}

383void mi_heap_destroy(mi_heap_t* heap) {
384 mi_assert(heap != NULL);
385 mi_assert(mi_heap_is_initialized(heap));
386 mi_assert(heap->no_reclaim);
387 mi_assert_expensive(mi_heap_is_valid(heap));
388 if (heap==NULL || !mi_heap_is_initialized(heap)) return;
389 #if MI_GUARDED
390 // _mi_warning_message("'mi_heap_destroy' called but MI_GUARDED is enabled -- using `mi_heap_delete` instead (heap at %p)\n", heap);
391 mi_heap_delete(heap);
392 return;
393 #else
394 if (!heap->no_reclaim) {
395 _mi_warning_message("'mi_heap_destroy' called but ignored as the heap was not created with 'allow_destroy' (heap at %p)\n", heap);
396 // don't free in case it may contain reclaimed pages
397 mi_heap_delete(heap);
398 }
399 else {
400 // track all blocks as freed
401 #if MI_TRACK_HEAP_DESTROY
402 mi_heap_visit_blocks(heap, true, mi_heap_track_block_free, NULL);
403 #endif
404 // free all pages
405 _mi_heap_destroy_pages(heap);
406 mi_heap_free(heap);
407 }
408 #endif
409}

411// forcefully destroy all heaps in the current thread
412void _mi_heap_unsafe_destroy_all(mi_heap_t* heap) {
413 mi_assert_internal(heap != NULL);
414 if (heap == NULL) return;
415 mi_heap_t* curr = heap->tld->heaps;
416 while (curr != NULL) {
417 mi_heap_t* next = curr->next;
418 if (curr->no_reclaim) {
419 mi_heap_destroy(curr);
420 }
421 else {
422 _mi_heap_destroy_pages(curr);
423 }
424 curr = next;
425 }
426}

432// Transfer the pages from one heap to the other
433static void mi_heap_absorb(mi_heap_t* heap, mi_heap_t* from) {
434 mi_assert_internal(heap!=NULL);
435 if (from==NULL || from->page_count == 0) return;
436
437 // reduce the size of the delayed frees
438 _mi_heap_delayed_free_partial(from);
439
440 // transfer all pages by appending the queues; this will set a new heap field
441 // so threads may do delayed frees in either heap for a while.
442 // note: appending waits for each page to not be in the `MI_DELAYED_FREEING` state
443 // so after this only the new heap will get delayed frees
444 for (size_t i = 0; i <= MI_BIN_FULL; i++) {
445 mi_page_queue_t* pq = &heap->pages[i];
446 mi_page_queue_t* append = &from->pages[i];
447 size_t pcount = _mi_page_queue_append(heap, pq, append);
448 heap->page_count += pcount;
449 from->page_count -= pcount;
450 }
451 mi_assert_internal(from->page_count == 0);
452
453 // and do outstanding delayed frees in the `from` heap
454 // note: be careful here as the `heap` field in all those pages no longer point to `from`,
455 // turns out to be ok as `_mi_heap_delayed_free` only visits the list and calls a
456 // the regular `_mi_free_delayed_block` which is safe.
457 _mi_heap_delayed_free_all(from);
458 #if !defined(_MSC_VER) || (_MSC_VER > 1900) // somehow the following line gives an error in VS2015, issue #353
459 mi_assert_internal(mi_atomic_load_ptr_relaxed(mi_block_t,&from->thread_delayed_free) == NULL);
460 #endif
461
462 // and reset the `from` heap
463 mi_heap_reset_pages(from);
464}

472// Safe delete a heap without freeing any still allocated blocks in that heap.
473void mi_heap_delete(mi_heap_t* heap)
474{
475 mi_assert(heap != NULL);
476 mi_assert(mi_heap_is_initialized(heap));
477 mi_assert_expensive(mi_heap_is_valid(heap));
478 if (heap==NULL || !mi_heap_is_initialized(heap)) return;
479
480 mi_heap_t* bheap = heap->tld->heap_backing;
481 if (bheap != heap && mi_heaps_are_compatible(bheap,heap)) {
482 // transfer still used pages to the backing heap
483 mi_heap_absorb(bheap, heap);
484 }
485 else {
486 // the backing heap abandons its pages
487 _mi_heap_collect_abandon(heap);
488 }
489 mi_assert_internal(heap->page_count==0);
490 mi_heap_free(heap);
491}

511static mi_heap_t* mi_heap_of_block(const void* p) {
512 if (p == NULL) return NULL;
513 mi_segment_t* segment = _mi_ptr_segment(p);
514 bool valid = (_mi_ptr_cookie(segment) == segment->cookie);
515 mi_assert_internal(valid);
516 if mi_unlikely(!valid) return NULL;
517 return mi_page_heap(_mi_segment_page_of(segment,p));
518}

521 mi_assert(heap != NULL);

523 return (heap == mi_heap_of_block(p));

527static bool mi_heap_page_check_owned(mi_heap_t* heap, mi_page_queue_t* pq, mi_page_t* page, void* p, void* vfound) {
528 MI_UNUSED(heap);
529 MI_UNUSED(pq);
530 bool* found = (bool*)vfound;
531 void* start = mi_page_start(page);
532 void* end = (uint8_t*)start + (page->capacity * mi_page_block_size(page));
533 *found = (p >= start && p < end);
534 return (!*found); // continue if not found
535}

543 return found;
544}

569static void mi_get_fast_divisor(size_t divisor, uint64_t* magic, size_t* shift) {
570 mi_assert_internal(divisor > 0 && divisor <= UINT32_MAX);
571 *shift = MI_SIZE_BITS - mi_clz(divisor - 1);
572 *magic = ((((uint64_t)1 << 32) * (((uint64_t)1 << *shift) - divisor)) / divisor + 1);
573}
574
575static size_t mi_fast_divide(size_t n, uint64_t magic, size_t shift) {
576 mi_assert_internal(n <= UINT32_MAX);
577 const uint64_t hi = ((uint64_t)n * magic) >> 32;
578 return (size_t)((hi + n) >> shift);
579}

581bool _mi_heap_area_visit_blocks(const mi_heap_area_t* area, mi_page_t* page, mi_block_visit_fun* visitor, void* arg) {
582 mi_assert(area != NULL);
583 if (area==NULL) return true;
584 mi_assert(page != NULL);
585 if (page == NULL) return true;
586
587 _mi_page_free_collect(page,true); // collect both thread_delayed and local_free
588 mi_assert_internal(page->local_free == NULL);
589 if (page->used == 0) return true;
590
591 size_t psize;
592 uint8_t* const pstart = _mi_segment_page_start(_mi_page_segment(page), page, &psize);
593 mi_heap_t* const heap = mi_page_heap(page);
594 const size_t bsize = mi_page_block_size(page);
595 const size_t ubsize = mi_page_usable_block_size(page); // without padding
596
597 // optimize page with one block
598 if (page->capacity == 1) {
599 mi_assert_internal(page->used == 1 && page->free == NULL);
600 return visitor(mi_page_heap(page), area, pstart, ubsize, arg);
601 }
602 mi_assert(bsize <= UINT32_MAX);
603
604 // optimize full pages
605 if (page->used == page->capacity) {
606 uint8_t* block = pstart;
607 for (size_t i = 0; i < page->capacity; i++) {
608 if (!visitor(heap, area, block, ubsize, arg)) return false;
609 block += bsize;
610 }
611 return true;
612 }
613
614 // create a bitmap of free blocks.
615 #define MI_MAX_BLOCKS (MI_SMALL_PAGE_SIZE / sizeof(void*))
616 uintptr_t free_map[MI_MAX_BLOCKS / MI_INTPTR_BITS];
617 const uintptr_t bmapsize = _mi_divide_up(page->capacity, MI_INTPTR_BITS);
618 memset(free_map, 0, bmapsize * sizeof(intptr_t));
619 if (page->capacity % MI_INTPTR_BITS != 0) {
620 // mark left-over bits at the end as free
621 size_t shift = (page->capacity % MI_INTPTR_BITS);
622 uintptr_t mask = (UINTPTR_MAX << shift);
623 free_map[bmapsize - 1] = mask;
624 }
625
626 // fast repeated division by the block size
627 uint64_t magic;
628 size_t shift;
629 mi_get_fast_divisor(bsize, &magic, &shift);
630
631 #if MI_DEBUG>1
632 size_t free_count = 0;
633 #endif
634 for (mi_block_t* block = page->free; block != NULL; block = mi_block_next(page, block)) {
635 #if MI_DEBUG>1
636 free_count++;
637 #endif
638 mi_assert_internal((uint8_t*)block >= pstart && (uint8_t*)block < (pstart + psize));
639 size_t offset = (uint8_t*)block - pstart;
640 mi_assert_internal(offset % bsize == 0);
641 mi_assert_internal(offset <= UINT32_MAX);
642 size_t blockidx = mi_fast_divide(offset, magic, shift);
643 mi_assert_internal(blockidx == offset / bsize);
644 mi_assert_internal(blockidx < MI_MAX_BLOCKS);
645 size_t bitidx = (blockidx / MI_INTPTR_BITS);
646 size_t bit = blockidx - (bitidx * MI_INTPTR_BITS);
647 free_map[bitidx] |= ((uintptr_t)1 << bit);
648 }
649 mi_assert_internal(page->capacity == (free_count + page->used));
650
651 // walk through all blocks skipping the free ones
652 #if MI_DEBUG>1
653 size_t used_count = 0;
654 #endif
655 uint8_t* block = pstart;
656 for (size_t i = 0; i < bmapsize; i++) {
657 if (free_map[i] == 0) {
658 // every block is in use
659 for (size_t j = 0; j < MI_INTPTR_BITS; j++) {
660 #if MI_DEBUG>1
661 used_count++;
662 #endif
663 if (!visitor(heap, area, block, ubsize, arg)) return false;
664 block += bsize;
665 }
666 }
667 else {
668 // visit the used blocks in the mask
669 uintptr_t m = ~free_map[i];
670 while (m != 0) {
671 #if MI_DEBUG>1
672 used_count++;
673 #endif
674 size_t bitidx = mi_ctz(m);
675 if (!visitor(heap, area, block + (bitidx * bsize), ubsize, arg)) return false;
676 m &= m - 1; // clear least significant bit
677 }
678 block += bsize * MI_INTPTR_BITS;
679 }
680 }
681 mi_assert_internal(page->used == used_count);
682 return true;
683}
