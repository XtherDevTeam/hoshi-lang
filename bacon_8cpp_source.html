19void bacon_flush_local_buffer() {
20 if (local_buffer.count == 0)
21 return;
22
23 auto new_node = (BufferNode *)mi_malloc(sizeof(BufferNode));
24 memcpy(new_node->entries, local_buffer.entries, sizeof(void *) * local_buffer.count);
25 if (local_buffer.count < ELYSIA_BACON_LOCAL_BUFFER_SIZE) {
26 new_node->entries[local_buffer.count] = nullptr;
27 }
28 local_buffer.count = 0;
29 do {
30 new_node->next = __atomic_load_n(&global_buffer_head, __ATOMIC_RELAXED);
31 } while (!__atomic_compare_exchange_n(&global_buffer_head,
32 &new_node->next,
33 new_node,
34 true,
35 __ATOMIC_RELEASE, // 使用 RELEASE 保证 memcpy 的数据对 GC 线程可见
36 __ATOMIC_RELAXED));
37 __atomic_fetch_add(&current_global_buffer_page_count, 1, __ATOMIC_RELEASE); // atomic operation to increment page count
38}
39void bacon_flush_global_buffer() {
40 // in STW mode, atomic operations are not needed
41 while (global_buffer_head != nullptr) {
42 auto next = global_buffer_head->next;
43 mi_free(global_buffer_head);
44 global_buffer_head = next;
45 }
46
47 current_global_buffer_page_count = 0; // reset page count
48}

137void bacon_stw() {
138 // stop-the-world
139 __atomic_store_n(&local_thread_should_sleep, 1, __ATOMIC_RELEASE);
140 while (__atomic_load_n(&bacon_current_active_threads, __ATOMIC_ACQUIRE) > 0)
141 CPU_RELAX();
142
143 // collect garbage
144 // mark grey
145 for (BufferNode *node = global_buffer_head; node != nullptr; node = node->next) {
146 for (int i = 0; i < ELYSIA_BACON_LOCAL_BUFFER_SIZE && node->entries[i]; i++) {
147 if (node->entries[i]->bacon_mark.get_color() == BaconMark::Color::Candidate)
148 bacon_mark_grey(node->entries[i]);
149 }
150 }
151
152 // scan again, find garbage
153 for (BufferNode *node = global_buffer_head; node != nullptr; node = node->next) {
154 for (int i = 0; i < ELYSIA_BACON_LOCAL_BUFFER_SIZE && node->entries[i]; i++) {
155 bacon_scan(node->entries[i]);
156 }
157 }
158
159 // collect white
160 SmallVector collected(1024);
161
162 for (BufferNode *node = global_buffer_head; node != nullptr; node = node->next) {
163 for (int i = 0; i < ELYSIA_BACON_LOCAL_BUFFER_SIZE && node->entries[i]; i++) {
164 node->entries[i]->bacon_mark.set_buffered(false);
165 bacon_collect_white(collected, node->entries[i]);
166 }
167 }
168
169 bacon_flush_global_buffer();
170
171 // wake up threads
172 __atomic_store_n(&local_thread_should_sleep, 0, __ATOMIC_RELEASE);
173 wake_all_on_address(&local_thread_should_sleep);
174
175 // now proceed finalization
176 for (int i = 0; i < collected.size; i++) {
177 auto &rtti_query = rtti_table[((YoiObject *)collected[i])->type_id];
178 if (rtti_query.finalizer) {
179 rtti_query.finalizer(collected[i]);
180 }
181 runtime_finalize_object((YoiObject *)collected[i]);
182 }
183
184 collected.finalize();
185}
